A finitely presented module is described by finitely many generators and relations.
These modules can be investigated by methods of linear algebra and play a central role in algebraic geometry.
When the base ring is coherent, this notion coincides with that of coherent modules, which is usually the generality to which one restricts classically.

Synthetically, however, our base ring $R$ is provably not coherent~\cite[Theorem~3.2.1]{topology-draft} and these two notions diverge.
As it turns out, neither is well-behaved in our setting.
The category of finitely presented $R$-modules lacks kernels, and the category of coherent $R$-modules, while always abelian, is much too small as it does not even contain crucial modules like $R^n$.
In this section, we study a larger category of finitely adequate modules~(\Cref{def:fin-adeq}) as a suitable replacement whose objects still retain a certain sense of finiteness.
This notion is not completely novel and has already been studied externally in~\cite[Chapter 06Z1]{stacks-project}.

\subsection{Quasi-Coherent and Adequate Modules}

Before defining finitely adequate modules~(\Cref{def:fin-adeq}), we shall first consider some more general notions.
In fact, the finitely adequate modules inherit several good properties from the broader class of adequate modules~(\Cref{def:adequate}), including tangentiality and weak quasi-coherence.
Using these properties, we prove that finitely copresented modules are projective among the adequate modules.
We begin by recounting the notion of quasi-coherence, introduced by Ingo Blechschmidt~\cite{ingo-thesis}.
Recall that for an $R$-module $M$ and a finitely presented $R$-algebra $A$, there is an evaluation map:
\begin{align*}
  \ev : M \otimes A & \rightarrow M^{\Spec A} \\
  m \otimes a & \mapsto (f \mapsto f(a)m)
\end{align*}

\begin{definition}\label{def:quasi-coherent}
  Let $M$ be an $R$-module.
  $M$ is \notion{quasi-coherent} if for every finitely presented $R$-algebra $A$, the map $\ev$ is an equivalence.
  We say that $M$ is \notion{weakly quasi-coherent} if this map is an equivalence for algebras of the form $R[r^{-1}]$, for each $r : R$.
  That is, the map $M[r^{-1}] \to M^{D(r)}$ sending $m/r^k$ to $p \mapsto m/r^k$
  (which is well-defined because $p$ witnesses that $r$ is invertible)
  is an equivalence.
\end{definition}

Typical examples of quasi-coherent modules include finitely presented $R$-algebras, considered as $R$-modules.
\rednote{Reference?}
Most notably for our purposes, the finite free modules $R^n$ are quasi-coherent for all $n : \N$.

\begin{remark}\label{rem:wqc-abelian}
  By \cite[Lemma 7.1.6 and Proposition 7.1.13]{draft}, weakly quasi-coherent $R$-modules are closed under taking kernels and cokernels.
  In particular, they form an abelian subcategory of $\modcat{R}$.
\end{remark}

It is not true that quasi-coherent modules are closed under kernels in general.
Thus, this collection of modules does not form an abelian category.
However, we can prove that they are closed under cokernels.

\begin{lemma}\label{lem:qc-colimits}
  Quasi-coherent $R$-modules are closed under cokernels.
\end{lemma}

\begin{proof}
  Let $M$ and $N$ be quasi-coherent $R$-modules, $f : M \to N$ a linear map, $C \defeq \Cok(f)$, and $p : N \twoheadrightarrow C$ the canonical projection.
  Our goal is to show that for any finitely presented $R$-algebra $A$, the map $\ev : C \otimes A \to C^{\Spec A}$ is an equivalence.
  The map $\ev$ is induced by the universal property of $C \otimes A$ as a cokernel.
  This is summarized by the following commutative diagram with exact rows:
  \begin{center}
    \begin{tikzcd}
      M \otimes A \ar{r}{f \otimes A} \ar{d}{\simeq} & N \otimes A \ar{d}{\simeq} \ar[two heads]{r}{p \otimes A} & C \otimes A \ar[dashed]{d}{\ev} \\
      M^{\Spec A} \ar{r}{f_\ast} & N^{\Spec A} \ar{r}{p_\ast} & C^{\Spec A}
    \end{tikzcd}
  \end{center}
  By the universal property of the cokernel, $\ev$ is an isomorphism if and only if $p_\ast$ is an epimorphism.
  Thus, it suffices to show that $p_\ast$ is an epimorphism.

  Let $K \defeq \Im(f)$ and $i : K \hookrightarrow N$ be the canonical inclusion.
  Now, $K$ is weakly quasi-coherent by \Cref{rem:wqc-abelian}.
  Considering the modules $K$, $N$, and $C$ as dependent modules over $A$, we get the following long exact sequence by taking cohomology:
  \begin{center}
    \begin{tikzcd}
      K^{\Spec A} \ar[hook]{r}{i_\ast} & N^{\Spec A} \ar{r}{p_\ast} & C^{\Spec A} \ar{r}{\partial} & \mathrm{H}^1(\Spec A, K)
    \end{tikzcd}
  \end{center}
  The map $p_\ast$ is an epimorphism, since $\mathrm{H}^1(\Spec A, K) = 0$ by \cite[Theorem 7.3.6]{draft}.
\end{proof}

\begin{remark}\label{rem:fp-is-qc}
  By \Cref{lem:qc-colimits}, every finitely presented $R$-module is quasi-coherent, as it is the cokernel of a map between finite free $R$-modules.
\end{remark}

Tangentiality of quasi-coherent modules can be proved by a direct computation.
By \Cref{cor:tangential-inf-lin}, we then get that quasi-coherent modules are infinitesimally linear.

\begin{lemma}\label{lem:disk-qc}
  Let $Q$ be a quasi-coherent $R$-module.
  A pointed map $(Q,0)^{\D(n)}$ is an $n$-variate linear monomial with coefficients in $Q$. More precisely, the map $Q^n \to (Q,0)^{\D(n)}$ sending $(q_1,\dots,q_n)$ to $(\epsilon_1,\dots,\epsilon_n) \mapsto \sum_{i=1}^n \epsilon_i q_i$ is an equivalence.
\end{lemma}

\begin{proof}
  By quasi-coherence, the map $Q^n \simeq Q \otimes \quot{R[x_1,\dots,x_n]}{\ideal{x_ix_j}} \to Q^{\D(n)}$ is an equivalence.
  We just need to check that maps with constant coefficient $0$ on the left correspond to maps such that $p(0)=0$ on the right.
  But indeed, given a $p:Q^{\D(n)}$ we have that the corresponding constant coefficient is $p(0)$.
\end{proof}

\begin{proposition}\label{prp:tangential-qc}
  Let $Q$ be a quasi-coherent $R$-module.
  Then $Q$ is tangential.
\end{proposition}

\begin{proof}
  Taking $n=1$ in \Cref{lem:disk-qc} gives that $\can : Q\to\tangent{Q}{0}$ is an equivalence.
\end{proof}

The category of quasi-coherent modules is far too small to be a faithful replacement for the algebro-geometric notion of a coherent sheaf of modules.
We therefore consider the category of adequate modules, which is the kernel closure of quasi-coherent modules.
As we will see, this larger family of modules will inherit tangentiality and weak quasi-coherence.

\begin{definition}\label{def:adequate}
  An $R$-module $M$ is \notion{adequate} if it is merely the kernel of a linear map $f : P \rightarrow Q$ where $P$ and $Q$ are quasi-coherent $R$-modules.
\end{definition}

Adequate $R$-modules enjoy the following properties as they form the kernel closure of quasi-coherent modules.

\begin{proposition}\label{prp:tangential-wqc-adeq}
  Adequate $R$-modules are tangential and weakly quasi-coherent. \earlyqed
\end{proposition}

We end this section by providing a class of projective adequate modules.
An $R$-module $L$ is called \notion{finitely copresented} if it is merely the kernel of a linear map $f : R^m \rightarrow R^n$.
The finitely copresented modules played an important role in \cite{diffgeo-article}, as the tangent spaces of schemes are always finitely copresented~\cite[Lemma 2.2.8]{diffgeo-article}.
Finitely copresented modules have also been studied externally in~\cite[Chapter 06Z1]{stacks-project}, where they go under the name of linearly adequate modules.

\begin{lemma}\label{lem:lin-adeq-lift}
  Let $L$ be a finitely copresented $R$-module.
  Every lifting problem
  \begin{center}
    \begin{tikzcd}
      & A \ar[two heads]{d}{e} \\
      L \ar{r}{v} & B
    \end{tikzcd}
  \end{center}
  merely has a linear solution when $A$ and $B$ are tangential and $M \defeq \Ker e$ is weakly quasi-coherent.
\end{lemma}

\rednote{Can we make the following proof nice?}

\begin{proof}
  By \cite[Theorem 7.3.6]{draft}, we have $H^1(L,M)=0$, since $L$, like any finitely copresented $R$-module, is an affine scheme. Indeed, it is the kernel of a linear map $f : R^m \to R^n$, and therefore the set of roots of a system of linear equations.

  By definition, we have $H^1(L,M)=||L \to K(M,1)||_0$ and therefore $L \to K(M,1)$ is connected.
  For $b : B$, we let $E_b$ denote the fiber of $e$ over $b$.
  Then $E_b$ is an $M$-torsor, or in other words an element of $K(M,1)$.
  So $E \circ v$ defines a map $L \to K(M,1)$, and we have $||E \circ v = (\argmhole \mapsto M)||$ by the connectedness noted above.
  This means that the dependent type $E \circ v$ merely has a section, since $(\argmhole \mapsto M)$ has a section.
  Let $h'' : L \rightarrow A$ denote the section after projecting onto $A$.
  The map $h' \defeq h'' - h''(0)$ is now a pointed map of type $(L,0) \pto (A,0)$.
  The map $h'$ still solves the problem as $e(h''(0)) = 0$.
  By \Cref{thm:linearize-lift}, there is a linear map $h : L \rightarrow A$ solving this lifting problem.
\end{proof}

\begin{remark}\label{rem:fcp-is-proj}
  Since adequate modules are tangential and weakly quasi-coherent, finitely copresented modules are projective in the category of adequate modules.
\end{remark}

\subsection{Finitely Adequate Modules Form an Abelian Category}

As discussed in the introduction, finding a faithful replacement for the algebro-geometric notion of a coherent sheaf of modules remains an open question.
Finitely presented $R$-modules do not form an abelian category \rednote{Reference?}, as this category is not closed under kernels.
This naturally leads us to consider the kernel closure of finitely presented $R$-modules as a potential replacement instead.
Even though it is known that this subcategory is not the exact replacement sought, it still exhibits many remarkable properties and forms an important first step in this program.
In this subsection, we will show that these modules, the finitely adequate modules~(\Cref{def:fin-adeq}), form an abelian subcategory of $\modcat{R}$.

\begin{definition}\label{def:fin-adeq}
  An $R$-module $M$ is \notion{finitely adequate} if it is merely the kernel of a homomorphism between finitely presented modules.
  We denote both the type and the category of finitely adequate modules by $\finadeq$\index{$\finadeq$}.
\end{definition}

Finitely adequate modules are in particular adequate modules, since finitely presented modules are quasi-coherent~(\Cref{rem:fp-is-qc}).
In this way, finitely adequate modules are tangential, infinitesimally linear, and weakly quasi-coherent~(\Cref{prp:tangential-wqc-adeq} and \Cref{cor:tangential-inf-lin}).

To show that $\finadeq$ forms an abelian category, we will first pass through a general construction modeled after Adelman's recognition theorem for free abelian categories~\cite[Theorem 1.15]{adelman-construction}.
Let $\mcal{B}$ be an abelian category and $\mcal{A}$ an additive subcategory $\mcal{A} \subseteq \mcal{B}$.
An object $M$ is called \notion{$\mcal{A}$-presented} if it is merely a cokernel of a morphism in $\mcal{A}$, and \notion{$\mcal{A}$-copresented} if it is merely a kernel of a morphism in $\mcal{A}$.
We say that an object $M : \mcal{B}$ is \notion{$\mcal{A}$-Adelman} if $M$ is merely the image of a morphism $f : L \to P$, where $L$ is $\mcal{A}$-copresented and $P$ is $\mcal{A}$-presented.
For these notions, the kernels and cokernels are taken in $\mcal{B}$, as we cannot ask for them to exist in a general additive subcategory $\mcal{A}$.
Let $\Ad{\mcal{A}}$ be the subcategory of $\mcal{B}$ consisting of all $\mcal{A}$-Adelman objects.
It is clear that $\Ad{\mcal{A}}$ is additive.

\begin{lemma}\label{lem:adel-desc}
  Suppose that the objects of $\mcal{A}$ are projective in $\Ad{\mcal{A}}$.
  Then, $M$ is merely the kernel of a morphism between $\mcal{A}$-presented objects if and only if $M$ is $\mcal{A}$-Adelman.
\end{lemma}

\begin{proof}
  In both directions, we are proving a mere proposition, so we may assume without loss of generality that the postulated maps are given explicitly.

  Assume first that $M$ is a kernel of $f : P \to Q$ with $P$ and $Q$ $\mcal{A}$-presented.
  The map $f$ lifts to the following square of objects in $\mcal{A}$
  \begin{center}
    \begin{tikzcd}
      & A_{p_2} \ar{r}{f_2} \ar{d}{p} & A_{q_2} \ar{d}{q} \\
      & A_{p_1} \ar{r}{f_1} \ar[two heads]{d} & A_{q_1} \ar[two heads]{d} \\
      M \ar[hook]{r} & P \ar{r}{f} & Q
    \end{tikzcd}
  \end{center}
  as the objects of $\mcal{A}$ are projective.
  Let $L$ be the kernel of the map $[f_1, -q] : A_{p_1} \oplus A_{q_2} \to A_{q_1}$, which is $\mcal{A}$-copresented.
  It is clear that the induced kernel map $L \twoheadrightarrow M$ is surjective, since the right hand square below
  \begin{center}
    \begin{tikzcd}
      L \ar[hook]{r} \ar[two heads, dashed]{d} & A_{p_1} \oplus A_{q_2} \ar{r}{[f_1, -q]} \ar[two heads]{d} \ar[phantom]{rd}[very near end]{\ulcorner} & A_{q_1} \ar[two heads]{d} \\
      M \ar[hook]{r} & P \ar{r}{f} & Q
    \end{tikzcd}
  \end{center}
  is a pushout.

  For the other direction, let $i: M \to P$ and $p: L \twoheadrightarrow M$ be the image factorization of $h : L \to P$.
  Furthermore, let $l : A_{l_1} \to A_{l_2}$ be a resolution of $L$, and $p : A_{p_2} \to A_{p_1}$ be a resolution of $P$.
  Setting $q \defeq l \oplus p : A_{l_1} \oplus A_{p_2} \to A_{l_2} \oplus A_{p_1}$ and $Q \defeq \Cok q$, we get an induced map $f : P \to Q$.
  It is clear that $M = \Ker f$.
\end{proof}

\begin{remark}\label{rem:adel-desc}
  Assuming instead that the objects of $\mcal{A}$ are injective yields a dual description: an object $M$ is $\mcal{A}$-Adelman if and only if it is merely the cokernel of a morphism between $\mcal{A}$-copresented objects.
\end{remark}

Before giving our construction, we will need the following lemma regarding kernels of maps in a commutative square:

\begin{lemma}\label{lem:diff-ker-is-ker}
  Suppose we have a commutative square:
  \begin{center}
    \begin{tikzcd}
      A \ar{r}{u} \ar{d}{f} & B \ar{d}{g} \\
      C \ar{r}{v} & D
    \end{tikzcd}
  \end{center}
  Then the kernel $K$ of the induced map $k_f : \Ker(u) \rightarrow \Ker(v)$ is equivalent to the kernel $K'$ of $(u, f) : A \to B \oplus C$.
\end{lemma}

\begin{proof}
  Firstly, $K'$ is given by the intersection $\Ker(u) \cap \Ker(f)$.
  By pullback pasting, we can make $K'$ the pullback of the following outer square:
  \begin{center}
    \begin{tikzcd}
      K' \ar[hook]{r} \ar[hook]{d} \ar[phantom]{dr}[very near start]{\lrcorner} & \Ker(f) \ar[hook]{d}{i_f} \ar{r} \ar[phantom]{dr}[very near start]{\lrcorner} & 0 \ar{d} \\
      \Ker(u) \ar[hook]{r}{i_u} & A \ar{r}{f} & C
    \end{tikzcd}
  \end{center}
  Secondly, the induced map $k_f$ sits in the commutative square
    \begin{center}
      \begin{tikzcd}
        \Ker(u) \ar[hook]{r}{i_u} \ar{d}{k_f} & A \ar{d}{f}\\
        \Ker(v) \ar[hook]{r}{i_v} & C
      \end{tikzcd}
    \end{center}
  Thus, $K'$ is the kernel of the composite $f \circ i_u = i_v \circ k_f$.
  Since $i_v$ is a monomorphism, we deduce that $K' = \Ker(k_f) = K$, as desired.
\end{proof}

\begin{proposition}\label{prp:ad-ab}
  Let $\mcal{B}$ be an abelian category and $\mcal{A} \subseteq \mcal{B}$ be an additive subcategory of $\mcal{B}$.
  $\Ad{\mcal{A}}$ is an abelian subcategory of $\mcal{B}$ if:
  \begin{itemize}[noitemsep]
    \item $\mcal{A}$-copresented objects are projective in $\Ad{\mcal{A}}$,
    \item $\mcal{A}$-presented objects are injective in $\Ad{\mcal{A}}$. 
    \earlyqed
  \end{itemize}
\end{proposition}

\begin{remark}\label{rem:rec-free-ab}
  Objects in $\mcal{A}$ are both projective and injective, since they are both $\mcal{A}$-presented and $\mcal{A}$-copresented.
\end{remark}

\begin{proof}
  We only need to verify that $\Ad{\mcal{A}}$ is closed under kernels and cokernels.
  Showing closure under kernels is sufficient, as the other case follows from a dual argument.
  
  Let $f : M \to N$ be a morphism between $\mcal{A}$-Adelman objects, and let $K$ be the kernel of $f$ taken in $\mcal{B}$.
  Since proving that $K$ is $\mcal{A}$-Adelman is a proposition, we can assume that there are resolutions $p : P_1 \to P_2$ and $q : Q_1 \to Q_2$, exhibiting $M$ and $N$ as kernels of $\mcal{A}$-presented objects respectively, by \Cref{lem:adel-desc}.
  Since $Q_1$ and $Q_2$ are injective in $\Ad{\mcal{A}}$, we can extend $f$ to $f_1 : P_1 \to Q_1$ and $f_2 : P_2 \to Q_2$ making the following diagram with exact rows commute:
  \begin{center}
    \begin{tikzcd}
      M \ar[hook]{r} \ar{d}{f} & P_1 \ar{d}{f_1} \ar{r}{p} & P_2 \ar{d}{f_2} \\
      N \ar[hook]{r} & Q_1 \ar{r}{q} & Q_2
    \end{tikzcd}
  \end{center}
  By \Cref{lem:diff-ker-is-ker}, $K$ is now the kernel of $(p, f_1) : P_1 \to P_2 \oplus Q_1$.
  Thus, $K$ is $\mcal{A}$-Adelman by \Cref{lem:adel-desc}.
\end{proof}

We now apply the above proposition to $\mcal{B} \defeq \modcat{R}$ and $\mcal{A} \defeq \finfree$.
Under this assignment, the $\finfree$-presented and $\finfree$-copresented objects are precisely the finitely presented and finitely copresented modules, which by \Cref{lem:adel-desc} yields $\Ad{\finfree} = \finadeq$.
Since finitely copresented modules are projective in $\adeq$, they are clearly projective in $\finadeq$ as well.
Therefore, it only remains to prove that finitely presented modules are injective in $\finadeq$.

This injectivity result will follow from duality.
Given an $R$-module $M$, we let $M^\ast \defeq \Hom_R(M,R)$ be the \notion{dual} of $M$.
We first recall that duality is an anti-equivalence between finitely copresented and finitely presented modules.

\begin{theorem}[{{\cite[Lemma 2.3.9, Corollary 2.3.10]{diffgeo-article}}}]\label{thm:lin-adeq-dual}
  Dualization restricts to an anti-equivalence $(\argmhole)^\ast : \fcpmod \rightarrow \fpmod$.
  Moreover, both finitely presented and finitely copresented modules are reflexive.
  \earlyqed
\end{theorem}

As stated, the theorem above is not strong enough to establish that finitely presented modules are injective among all finitely adequate ones. 
By extending this duality result to all finitely adequate modules, we obtain the desired injectivity result.

\begin{lemma}\label{lem:dual-exact-fp-inc}
  Let $M$ be a finitely adequate $R$-module, and $P$ a finitely presented $R$-module.
  Any lifting problem
  \begin{center}
    \begin{tikzcd}
      M \ar[hook]{d}{m} \ar{r}{u} & R \\
      P
    \end{tikzcd}
  \end{center}
  merely admits a linear solution.
\end{lemma}

\rednote{The following proof broke a bit when I removed homology of composable pairs. This shouldn't be too much of a problem, if we can adapt Fabian's new ideas into this paper.}

\begin{proof}
  Since we're proving a proposition, we can choose $p : L \twoheadrightarrow M$ with $L$ finitely copresented.
  By the proof of \Cref{lem:adel-desc}, we get a pair of composable morphisms $f : R^l \to R^m$ and $g : R^m \to R^n$ such that $L = \Ker g$ and $P = \Cok f$.
  This gives us the following commutative diagram, in which every row and column is exact:

  \begin{center}
    \begin{tikzcd}
      K \ar{d}{} \ar[hook]{r} & R^{l} \ar{r}{g \circ f} \ar{d}{f} & R^{n} \ar[equal]{d} \\
      L \ar[two heads]{d}{p} \ar[hook]{r} & R^{m} \ar{r}{g} \ar[two heads]{d} & R^{n} \\
      M \ar[hook]{r}{m} & P
    \end{tikzcd}
  \end{center}

  We now proceed with a diagram chase argument to show that $u : M \to R$ extends along $m :M \hookrightarrow P$.
  The linear map $u : M \to R$ restricts to a map $l_1 : L \to R$ that is zero on $K$.
  By~\cite[Lemma 2.3.7]{diffgeo-article}, it is possible to extend $l_1$ to $l_2 : R^{m} \to R$.
  $l_2$ will descend to $M$ if and only if it vanishes on the image of $f$.
  Although this is not generally the case, we can construct a linear correction $c : R^{m} \to R$ such that $l_2 - c$ has this property.

  To construct $c$, let $c_0 \defeq l_2 \circ f$.
  Note that $c_0$ vanishes on $K$ and therefore descends to a linear map $c_1 : \Im(g \circ f) \to R$.
  By \cite[Lemma 2.3.8]{diffgeo-article}, $c_1$ extends to a map $c_2 : R^{n} \to R$.
  Now, $c \defeq c_2 \circ g \circ f$ is a linear map on $R^{l}$ that is zero on $K$,
  so $l_2 - c$ vanishes on the image of $f$.
  Therefore, $l_2 - c$ descends to a map $l_3 : P \to R$.

  Finally, note that $l_3$ restricts to $l_1$ on $L$.
  Thus, we have $u = l_3 \circ m$.
\end{proof}

\begin{proposition}
  Let $M$ be a finitely adequate $R$-module. Then $M^\ast$ is finitely adequate.
\end{proposition}

\begin{proof}
  By \Cref{lem:adel-desc}, there merely exists a surjection $p : L \twoheadrightarrow M$ with $L$ finitely copresented and an injection $i : M \hookrightarrow P$ with $P$ finitely presented.
  Taking duals, we get that $p^\ast : M^\ast \hookrightarrow L^\ast$ is an injection, and $i^\ast : P^\ast \twoheadrightarrow M^\ast$ is a surjection by \Cref{lem:dual-exact-fp-inc}.
  Thus, $M^\ast$ is finitely adequate by \Cref{lem:adel-desc}.
\end{proof}

\begin{proposition}\label{prp:can-dual-iso}
  Let $M$ be a finitely adequate $R$-module. Then $M$ is reflexive.
\end{proposition}

\begin{proof}
  By \Cref{lem:adel-desc}, there merely exists a surjection $p : L \twoheadrightarrow M$ with $L$ finitely copresented and an injection $i : M \hookrightarrow P$ with $P$ finitely presented. Let $f \defeq i \circ p : L \to P$ denote the composite of these maps.
  Taking double duals, we get that $p^{\ast\ast} : M^{\ast\ast} \twoheadrightarrow L^{\ast\ast}$ is a surjection, and $i^{\ast\ast} : P^{\ast\ast} \hookrightarrow M^{\ast\ast}$ is an injection by \Cref{lem:dual-exact-fp-inc}.
  We thus get that $M^{\ast\ast} = \Im (f^{\ast\ast})$.
  By \Cref{thm:lin-adeq-dual}, $\ev$ yields an isomorphism between $f$ and $f^{\ast\ast}$ as $L$ is finitely copresented and $P$ is finitely presented.
  Thus, $M = \Im f = \Im (f^{\ast\ast}) = M^{\ast\ast}$.
\end{proof}

Combining the above propositions, we get the following result:

\begin{corollary}\label{cor:dual-equiv}
  Taking duals $(\argmhole)^{\ast} : \finadeq \rightarrow \finadeq$ is an anti-equivalence of categories. \earlyqed
\end{corollary}

With the improved duality result, we can now dualize \Cref{rem:fcp-is-proj} in the subcategory of finitely adequate modules.

\begin{proposition}\label{prp:fp-is-inj}
  Finitely presented modules are injective in $\finadeq$.
\end{proposition}

\begin{proof}
  By \Cref{cor:dual-equiv} it is sufficient to prove that finitely copresented modules are projective in $\finadeq$.
  This is precisely \Cref{rem:fcp-is-proj}.
\end{proof}

\begin{theorem}\label{thm:fin-adeq-abelian}
  The category $\finadeq$ is an abelian subcategory of $\modcat{R}$, i.e. $\finadeq$ is closed under kernels and cokernels.
\end{theorem}

\begin{proof}
  This is a direct consequence of \Cref{prp:ad-ab}, \Cref{rem:fcp-is-proj}, and \Cref{prp:fp-is-inj}.
\end{proof}

\begin{remark}\label{rem:free-abelian}
  By \cite[Theorem 1.15]{adelman-construction}, $\finadeq$ is the free abelian category over $\finfree$.
  More explicitly, it has the following universal property:
  For every abelian category $\mathcal{B}$ and every additive functor $F:\finfree \to \mathcal{B}$, there exists a unique exact functor $K$ as follows:
  \begin{center}
    \begin{tikzcd}
      \finfree \ar{d} \ar{r}{F} & \mathcal{B} \\
      \finadeq \ar[dashed]{ru}[swap]{K}
    \end{tikzcd}
  \end{center}
  Consequently, the data needed to specify an exact functor $K$ is an assignment of $R$ to an object $KR$ in $\mathcal{B}$ and a group homomorphism $\Hom_R(R,R) \rightarrow \mathcal{B}(KR,KR)$.
\end{remark}

Since $\finadeq$ is the free abelian category over $\finfree$, it follows by \cite[Proposition 1.3]{adelman-construction} that finitely copresented modules are closed under kernels and finitely presented modules are closed under cokernels.
Consequently, both classes are closed under direct summands.
We get the following proposition as a result.

\begin{proposition}\label{prp:proj-inj-char}
  Let $M$ be finitely adequate. $M$ is injective if and only if $M$ is finitely presented, and $M$ is projective if and only if $M$ is finitely copresented.
  \earlyqed
\end{proposition}

\begin{proposition}
  Let $A$ be a closed submodule of a finitely adequate module $M$. Then $A$ is finitely adequate if $A$ is tangential.
\end{proposition}

\begin{proof}
  Let $M$ be presented by the finitely copresented modules $K \to L \twoheadrightarrow M$. Consider the pullback $L' \coloneq A \times_M L$.
  Since pullbacks of $R$-modules are taken as pullbacks on the underlying sets, we have that $L'$ is a closed submodule of $L$, and is therefore a scheme by \cite[Proposition 5.4.3]{draft}.
  Therefore, the tangent space $\tangent{L'}{0}$ is a finitely copresented module by \cite[Lemma 2.2.8]{diffgeo-article}.
  Since tangential modules are closed under limits \Cref{prp:op-closure-inf-lin-tangential}, $L'$ is also tangential, and therefore also finitely copresented.
  The sequence $K \to L' \twoheadrightarrow A$ presents $A$ as a cokernel of a map of finitely copresented modules, so $A$ is finitely adequate.
\end{proof}

\begin{remark}
  The functor $\hood{\argmhole}{1}$ of the previous proposition is full when restricted even further to finitely copresented $R$-modules.
  This, however, already fails on finitely adequate $R$-modules.

  To see this, consider the modules $(\varepsilon)$ for every $\varepsilon:\D(1)$. We have that $(\varepsilon) = \hood{(\varepsilon),0}{1}$, so that the functor $\hood{\argmhole}{1}$ acts as the identity on maps between such modules. So if $\hood{\argmhole}{1}$ is full on finitely adequate modules, then any pointed map from $((\varepsilon),0)$ to itself is linear. Let us consider the maps $\alpha_\varepsilon : (\varepsilon) \to (\varepsilon), r\varepsilon \mapsto r^2\varepsilon$ for every $\varepsilon:\D(1)$. Finitely adequate modules form a weak Serre subcategory of $\modcat{R}$, meaning that finitely adequate modules are stable under extension. We need two auxiliary results.

\begin{lemma}\label{lem:fin-adeq-ext-pb-red}
Suppose we have a pullback diagram:
  \begin{center}
    \begin{tikzcd}
     p^*E \ar{r} \ar{d} \ar[phantom]{rd}[very near start]{\lrcorner} & A \ar[two heads]{d}{p} \\
      E \ar{r} & B
    \end{tikzcd}
  \end{center}
  where $A$ and $B$ are finitely adequate. Then $E$ is finitely adequate if and only if $p^*E$ is finitely adequate.
\end{lemma}

\begin{proof}
  Suppose that $E$ is finitely adequate.
  Since $p^*E$ is a pullback of finitely adequate $R$-modules, it is itself finitely adequate.

  Now assume that $p^*E$ is finitely adequate.
  As a pullback of a surjective map, the map $p^*E\to E$ is surjective, and its kernel is equivalent to $\Ker(p)$ by pullback pasting.
  $\Ker(p)$ is finitely adequate by \Cref{thm:fin-adeq-abelian}, and hence $E$ is finitely adequate by \Cref{thm:fin-adeq-abelian} as well.
\end{proof}

\begin{lemma}\label{prp:extension-fin-adeq-lin-adeq}
Given $B$ finitely adequate and $L$ finitely copresented, any extension:
    \begin{center}
    \begin{tikzcd}
      B \ar[hook]{r} & E \ar[two heads]{r} & L
    \end{tikzcd}
  \end{center}
   splits, so that $E$ is finitely adequate.
\end{lemma}

\begin{proof}
Finding a splitting amounts to showing that any lifting problem:
  \begin{center}
    \begin{tikzcd}
      & E \ar[two heads]{d}{p} \\
      L \ar[equal]{r} & L
    \end{tikzcd}
  \end{center}
  where $L$ is finitely adequate and $B \defeq \Ker(p)$ is finitely adequate, merely has a linear solution.
  Observe that $E$ is tangential by \Cref{prp:op-closure-inf-lin-tangential}, since $B$ and $L$ are tangential by \Cref{prp:tangential-wqc-adeq}.
  $B$ is also weakly quasi-coherent by \Cref{prp:tangential-wqc-adeq}.
  Thus, the lifting problem has a solution by \Cref{lem:lin-adeq-lift}. Once we have a splitting, we get $E=B \oplus L$ which is indeed finitely adequate.
\end{proof}

\begin{theorem}\label{thm:fin-adeq-ext-closed}
  Finitely adequate $R$-modules are stable under extension.
\end{theorem}

\begin{proof}
 Consider an extension problem:
  \begin{center}
    \begin{tikzcd}
      B \ar[hook]{r} & E \ar[two heads]{r} & A
    \end{tikzcd}
  \end{center}
  where $A$ and $B$ are finitely adequate. We merely get a resolution $L' \rightarrow L \overset{p}{\twoheadrightarrow} A$ with $L$ and $L'$ finitely copresented. Consider the pullback square:
    \begin{center}
    \begin{tikzcd}
     p^*E \ar{r} \ar{d} \ar[phantom]{rd}[very near start]{\lrcorner} & L \ar[two heads]{d}{p} \\
      E \ar{r} & A
    \end{tikzcd}
  \end{center}
  By \Cref{lem:fin-adeq-ext-pb-red} it is enough to show that $p^*E$ is finitely adequate, but the kernel of the top map is equal to $B$, so we have a new extension problem:
    \begin{center}
    \begin{tikzcd}
      B \ar[hook]{r} & p^*E \ar[two heads]{r} & L
    \end{tikzcd}
  \end{center}
 We can conclude by \Cref{prp:extension-fin-adeq-lin-adeq}.
\end{proof}


We finish this subsection by characterizing finitely presented $R$-modules amongst all finitely adequate modules as exactly those that are quasi-coherent.
For $R$-modules $M$ and $N$, we will make use of the $R$-module homomorphism
  \[
    \beta_{M,N} : M \otimes N \to \Hom_R(M^\ast, N), \quad
    m \otimes n \mapsto (\lambda \mapsto \lambda(m)n)
  \]
This homomorphism is natural in $M$ and $N$.

\begin{lemma}
  \label{identification-hom-tensor-for-fp-algebras}
  Let $M$ be a reflexive $R$-module.
  Then $M$ is quasi-coherent if and only if for every finitely presented $R$-algebra $A$ the map
  $\beta_{M,A}$ is an equivalence.
  Here $A$ is regarded as an $R$-module.
\end{lemma}

\begin{proof}
  Suppose $M$ is reflexive.
  We consider the following composite map:
  \begin{align*}
    M \otimes A &\longrightarrow M^{\Spec A} \\
                &\overset{\simeq}{\longrightarrow} (M^{\ast\ast})^{\Spec A} \\
                &\overset{\simeq}{\longrightarrow} \Hom_R(M^\ast, R^{\Spec A}) \\
                &\overset{\simeq}{\longleftarrow} \Hom_R(M^\ast, A)
  \end{align*}
  The first map is the one from the definition of quasi-coherence.
  The second map is an equivalence due to reflexivity of $M$.
  The third map comes from currying on the level of maps between types and extends to maps with the appropriate linearity condition, since $R^{\Spec A}$ carries the pointwise module structure.
  The last map is an equivalence by duality.
  We omit the verification that this composite agrees with the map $\beta_{M,A}$.
  Thus, the first map is an equivalence if and only if the composite is an equivalence.
\end{proof}

\begin{corollary}
  \label{identification-hom-tensor-for-fp-modules}
  Let $M$ be a reflexive and quasi-coherent $R$-module.
  Then for every finitely presented $R$-module $N$ the map $\beta_{M,N}$
  is an equivalence.
\end{corollary}

\begin{proof}
  Consider the trivial square-zero extension $R \oplus N$ of $N$, which is a finitely presented $R$-algebra and fits into an exact sequence
  \[
    0 \longrightarrow N \longrightarrow R \oplus N \overset{\pi}{\longrightarrow} R \longrightarrow 0\rlap{.}
  \]
  This sequence is split exact as a sequence of $R$-modules, hence any additive functor will preserve its exactness.
  Therefore, we get a commutative diagram
  \[
  \begin{tikzcd}[column sep=1.5cm]
    0 \ar[d] & 0 \ar[d] \\
    M \otimes N \ar[d]\ar[r, "\beta_{M,N}"] & \Hom_R(M^\ast,N) \ar[d] \\
    M\otimes (R\oplus N) \ar[r,"\beta_{M,R \oplus N}"] \ar[d,"M \otimes \pi"'] & \operatorname{Hom}_R(M^\ast,R\oplus N) \ar[d,"\pi_\ast"] \\
    M \otimes R \ar[d]\ar[r,"\beta_{M,R}"] & \operatorname{Hom}_R(M^\ast,R) \ar[d] \\
    0 & 0
  \end{tikzcd}
\]
with exact columns.
The map $\beta_{M,R \oplus N}$ is an equivalence by \Cref{identification-hom-tensor-for-fp-algebras} and the map $\beta_{M,R}$ is an equivalence by the reflexivity of $M$.
By the universality of kernels we then see that $\beta_{M,N}$ is an equivalence as well.
\end{proof}

\begin{remark}
  The equivalence of \Cref{identification-hom-tensor-for-fp-modules} does not generalize to all finitely adequate $R$-modules $N$.
  If $\beta_{M,N}$ is an equivalence for all finitely adequate $N$,
  then since $M \otimes (\argmhole)$ is right exact and $\Hom_R(M^\ast,\argmhole)$ is left exact,
  we deduce that both would be exact functors on the subcategory of finitely adequate modules.
  We will show that $M \otimes (\argmhole)$ is not exact, even if $M$ is finitely presented.
  (Note that in this case $\Hom_R(M^\ast,\argmhole)$ \emph{is} exact, since $M^\ast$ is a projective module
  when $M$ is finitely presented.)

  Now let us assume that for all $r:R$ the functor $R/(r)\otimes(\argmhole)$ is exact on finitely adequate modules. The monomorphism $(r) \lhook\joinrel\to R$ induces the zero map $R/(r) \otimes (r) \to R/(r) \otimes R$, so by exactness we get $R/(r) \otimes (r) = (r)/(r^2) = 0$, which implies the mere existence of an $a:R$ such that $r=ar^2$. Since $R$ is local we have $(ar \neq 0) \lor (1-ar \neq 0)$ and a simple case analysis give $(r=0) \lor (r \neq 0)$, which cannot hold for all $r:R$.
\end{remark}

\begin{proposition}
  Let $M$ be an $R$-module that is finitely adequate and quasi-coherent.
  Then $M$ is finitely presented.
\end{proposition}

\begin{proof}
  We show that $M^\ast$ is finitely copresented. 
  By \Cref{prp:proj-inj-char}, this is equivalent to showing that $M^\ast$ is projective, i.e.\ we need to show that $H :\equiv \Hom_R(M^\ast,\argmhole)$ is exact on the category of finitely adequate modules.
  For this let $0 \to A \to B \to C \to 0$ be an exact sequence of finitely adequate $R$-modules.
  We get an induced exact sequence $0 \to H(A) \to H(B) \to H(C) \to \Ext_R^1(M^\ast,A)$, where $\Ext_R^1$ is defined as in \cite[Definition 2.1.5]{ext-groups-in-hott}.
  Since finitely adequate $R$-modules are closed under extensions, $\Ext_R^1$ taken either in this category or the category of all $R$-modules is the same.
  Then the proof of \cite[Proposition 2.5.4]{ext-groups-in-hott} also shows that $\Ext_R^1(M^\ast,A)$ can be calculated using an injective resolution of $A$ in the category of finitely adequate $R$-modules.
  Recall that the injective modules are exactly the finitely presented modules.
  Thus, we have $\Ext_R^1(M^\ast,A)=0$, since \Cref{identification-hom-tensor-for-fp-modules} implies that $\Hom_R(M^{\ast},\argmhole)$ is exact on the category of finitely presented modules.
\end{proof}

\subsection{The Closed Symmetric Monoidal Structure}

In this subsection, we show that the category $\finadeq$ of finitely adequate $R$-modules carries a closed symmetric monoidal structure.
To construct this structure, we first establish that $\finadeq$ is closed under taking homs.
Namely, we first want to show that the module of homomorphisms between any two finitely adequate modules is itself finitely adequate.
We will start with the following lemmas.

\begin{lemma}\label{lem:fp-tensor-fp}
  Let $M$ and $N$ be finitely presented $R$-modules.
  Then the module $M \otimes N$ is also finitely presented.
\end{lemma}

\begin{proof}
  Since we want to prove a proposition, we can let $p : R^{m_2} \to R^{m_1}$ be a linear map such that $M = \Cok(p)$.
  By taking tensors we get that $M \otimes N$ is the cokernel of $p \otimes N : R^{m_2} \otimes N \to R^{m_1} \otimes N$.
  The domain and codomain of $p \otimes N$ are finitely presented, so $M \otimes N$ is finitely presented as well.
\end{proof}

\begin{lemma}\label{lem:lin-adeq-fp-hom-fp}
  Let $L$ be finitely copresented and $P$ be finitely presented.
  Then the module $\Hom_R(L,P)$ is finitely presented.
\end{lemma}

\begin{proof}
  By \Cref{identification-hom-tensor-for-fp-modules}, we have the following chain of equivalences:
  \begin{equation*}
    \Hom_R(L,P) \cong \Hom_R(L^{\ast\ast},P) \cong L^\ast \otimes P\textup{.}
  \end{equation*}
  By \Cref{lem:fp-tensor-fp}, we get that $\Hom_R(L,P)$ is finitely presented, as $L^\ast$ and $P$ are so.
\end{proof}

\begin{proposition}\label{prp:fin-adeq-hom-closure}
  Let $M$ and $N$ be finitely adequate modules.
  Then the module $\Hom_R(M,N)$ is finitely adequate as well.
\end{proposition}

\begin{proof}
  Since we're proving a proposition, we can let $l : L_2 \to L_1$ be a resolution of $M$, and $p : P_1 \to P_2$ be a resolution of $N$.
  Since hom is left-exact, we get the following diagram with exact rows and columns:
  \begin{center}
    \begin{tikzcd}
      & \Hom_R(L_1,P_2) \ar{r}{l^\ast} & \Hom_R(L_2,P_2) \\
      & \Hom_R(L_1,P_1) \ar{r}{l^\ast} \ar{u}{p_\ast} & \Hom_R(L_2,P_1) \ar{u}{p_\ast} \\
      \Hom_R(M,N) \ar[hook]{r} & \Hom_R(L_1,N) \ar{r}{l^\ast} \ar[hook]{u} & \Hom_R(L_2,N) \ar[hook]{u}
    \end{tikzcd}
  \end{center}

  Now, by \Cref{lem:diff-ker-is-ker}, we see that $\Hom_R(M,N)$ is the kernel of
  \[
    (p_\ast, l^\ast) : \Hom_R(L_1,P_1) \to \Hom_R(L_1,P_2) \oplus \Hom_R(L_2,P_1).
  \]
  The modules $\Hom_R(L_i,P_j)$ are finitely presented by \Cref{lem:lin-adeq-fp-hom-fp} for all $i,j \in \{1,2\}$.
  Thus, $\Hom_R(M,N)$ is finitely adequate as it is the kernel of a map between finitely presented modules.
\end{proof}

Our next goal is to show that for each finitely adequate $M$, the functor $\Hom_R(M, \argmhole) : \finadeq \to \finadeq$
has a left adjoint.
A natural guess would be the tensor product functor $\argmhole \otimes M$.
However, even the tensor product $N \otimes M$ of finitely copresented modules fails to be finitely adequate in general.  (See \Cref{cex:tensor-not-closed}.)
In contrast, if $M$ is finitely presented, $N \otimes M$ is the cokernel of a map between finite direct sums of $N$, and is therefore finitely adequate when $N$ is. (See \Cref{lem:fp-tensor-fp} for the case where both are finitely presented.)
A symmetrical argument works when $N$ is finitely presented.

Despite this, $\Hom_R(M, \argmhole)$ still admits a left adjoint in $\finadeq$.
We will show that this adjoint is given by the \notion{double-dual tensor product}
\[
  M \dbldualtensor N \defeq (M \otimes N)^{\ast\ast}.
\]
By the standard hom-tensor adjunction, there is already a natural isomorphism $\Hom_R(A, \Hom_R(B, C)) \simeq \Hom_R(A \otimes B, C)$.
Thus, proving that $(-\dbldualtensor-)$ is the internal left adjoint reduces to verifying that given $A$ and $B$ finitely adequate, we have that $A\dbldualtensor B$ is finitely adequate and that the canonical evaluation map $\ev_{A \otimes B} : A \otimes B \to A \dbldualtensor B$ induces an isomorphism
\[
  \ev_{A \otimes B}^\ast : \Hom_R(A \dbldualtensor B, C) \xrightarrow{\;\simeq\;} \Hom_R(A \otimes B, C)
\]
for each finitely adequate module $C$.
The composite of these natural isomorphisms yields the desired natural isomorphism $\Hom_R(A \dbldualtensor B, C) \simeq \Hom_R(A, \Hom_R(B, C))$.
It is clear that $\ev_{A \otimes B}^\ast$ is natural in all three arguments.
Now, $\ev_{A \otimes B}^\ast$ is an isomorphism precisely when every morphism $f : A \otimes B \to C$ factors uniquely through $\ev_{A \otimes B} : A \otimes B \to A \dbldualtensor B$.

We first carry out this factorization in the general setting of monads.
Let $T$ be a monad on $\mathcal{C}$ with multiplication $\mu$ and unit $\eta$.
In our situation, $\mathcal{C}$ will be the category of $R$-modules and $T$
will be the double-dual operation.

\begin{definition}\label{def:monad-reflexive}
  Let $X$ be an object of $\mathcal{C}$. We say that $X$ is:
  \begin{itemize}
    \item \notion{$T$-torsionless} if the unit $\eta_X$ is a monomorphism,
    \item \notion{$T$-reflexive} if the unit $\eta_X$ is an isomorphism.
  \end{itemize}
\end{definition}

\begin{lemma}\label{lem:monad-app-torsionless}
  Let $X$ be an object of $\mathcal{C}$. Then $TX$ is $T$-torsionless.
\end{lemma}

\begin{proof}
  By unitality, we have $\mu_X \circ \eta_{TX} = \id_{TX}$, so $\eta_{TX}$ is a split monomorphism.
\end{proof}

\begin{definition}\label{def:rel-epi}
  Let $\mathfrak{C}$ be a class of objects in $\mathcal{C}$. A map $f : X \to Y$ is \textbf{$\mathfrak{C}$-epic} if for every $Z$ in $\mathfrak{C}$ and all morphisms $g, h : Y \to Z$ such that $g \circ f = h \circ f$, we have $g = h$.
\end{definition}

The monad $T$ is said to be \notion{well-pointed} if $T\eta_X = \eta_{TX}$ for every object $X$ of $\mathcal{C}$.
We will not discuss well-pointed monads in this global sense, but rather focus on the local notion.
For a treatment of well-pointed monads, we refer the reader to Chapter 4.2 of Borceux's book~\cite{Borceux_1994_II}.
The next result is similar to Proposition 4.2.3 of Borceux~\cite{Borceux_1994_II}, but we do not assume that $T$ is well-pointed, which forces us to give a different proof.

\begin{proposition}\label{prp:monad-well-pointedness-torsionless-epi}
  Let $X$ be an object of $\mathcal{C}$. The following are equivalent:
  \begin{enumerate}[(i), noitemsep]
    \item $TX$ is $T$-reflexive,
    \item $T\eta_X = \eta_{TX}$,
    \item $\eta_X$ is $T$-torsionless-epic.
  \end{enumerate}
\end{proposition}

\begin{proof}
  (i) $\Rightarrow$ (iii): Assume that $TX$ is $T$-reflexive, i.e., that $\eta_{TX}$ is an isomorphism.
  Then $\mu_X$ is an isomorphism, since it is a retraction of $\eta_{TX}$,
  and $T\eta_X$ is an isomorphism, since it is a section of $\mu_X$.
  Consider two morphisms $f, g : TX \to Y$, where $Y$ is $T$-torsionless and $f \circ \eta_X = g \circ \eta_X$.
  By applying $T$ to this equality, we obtain $Tf \circ T\eta_X = Tg \circ T\eta_X$.
  Thus, $Tf = Tg$ since $T\eta_X$ is an isomorphism.
  By naturality of $\eta$, we have $\eta_Y \circ f = Tf \circ \eta_{TX} = Tg \circ \eta_{TX} = \eta_Y \circ g$.
  Since $Y$ is $T$-torsionless, $\eta_Y$ is a monomorphism and thus $f = g$.

  (iii) $\Rightarrow$ (ii):  Assume that $\eta_X$ is $T$-torsionless-epic.
  By naturality of $\eta$, we have $\eta_{TX} \circ \eta_X = T\eta_X \circ \eta_X$.
  Since $TX$ is $T$-torsionless by \Cref{lem:monad-app-torsionless}, we obtain $T\eta_X = \eta_{TX}$.

  (ii) $\Rightarrow$ (i): Assume that $T\eta_X = \eta_{TX}$.
  Our goal is to show that $\eta_{TX}$ is an isomorphism, so it suffices to show that $T\eta_X$ is an isomorphism.
  By unitality, we have that $\mu_X \circ T\eta_X = \id_{TX}$, so it remains to check the other composite.
  By unitality and using $T\eta_X = \eta_{TX}$, we have
  \[
    \id_{T^2X} = \mu_{TX} \circ T\eta_{TX} = \mu_{TX} \circ T^2\eta_{X} .
  \]
  From the naturality square of $\mu$ applied to $\eta_X$, we get the following commutative square:
  \begin{center}
    \begin{tikzcd}
      T^2X \ar{d}{\mu_X} \ar{r}{T^2\eta_{X}} & T^3X \ar{d}{\mu_{TX}} \\
      TX \ar{r}{T\eta_{X}} & T^2X
    \end{tikzcd}
  \end{center}
  Combining these two facts, we deduce that $T\eta_X \circ \mu_X = \id_{T^2X}$, so $T\eta_X$ is an isomorphism.
\end{proof}


\begin{proposition}\label{prp:monad-factorization}
  Let $f : X \to Y$ be a morphism in $\mathcal{C}$. Then $f$ factors uniquely through $\eta_X$ if both $TX$ and $Y$ are $T$-reflexive.
\end{proposition}

\begin{proof}
  The naturality square of $\eta$ applied to $f$
\begin{center}
  \begin{tikzcd}
    X \ar{r}{f} \ar{d}{\eta_X} & Y \ar{d}{\eta_Y}[left]{\simeq} \\
    T X \ar{r}{T f} & T Y
  \end{tikzcd}
\end{center}
  yields a factorization $f = \eta_Y^{-1} \circ T f \circ \eta_X$ of $f$ through $\eta_X$.
  For uniqueness, assume we are given $f', f'' : TX \to Y$ such that $f' \circ \eta_X = f'' \circ \eta_X$.
  By \Cref{prp:monad-well-pointedness-torsionless-epi}, $\eta_X$ is $T$-torsionless-epic, which yields $f' = f''$.
\end{proof}

Now we specialize to a monad coming from an adjoint pair
$F \dashv G$ with $F : \mathcal{C} \to \mathcal{D}$, $G : \mathcal{D} \to \mathcal{C}$, unit $\eta : \id_{\mathcal{C}} \to GF$ and counit $\varepsilon : FG \to \id_{\mathcal{D}}$.
The associated monad is $T \coloneq GF$, with multiplication $\mu \coloneq G \varepsilon_F$ and unit $\eta$.
In the case where the monad is induced from an adjunction, we can make a slight improvement to \Cref{prp:monad-well-pointedness-torsionless-epi}.

\begin{proposition}\label{prp:monad-adjunction-well-pointed-improvement}
  Let $X$ be an object of $\mathcal{C}$.
  Then $GF\eta_X = \eta_{GFX}$ if and only if $F\eta_X$ is an isomorphism.
\end{proposition}

\begin{proof}
  Assume first that $GF\eta_X = \eta_{GFX}$.
  This part of the proof is completely analogous to the implication (ii) $\Rightarrow$ (i) in \Cref{prp:monad-well-pointedness-torsionless-epi}, but we write it out for completeness.
  From the triangle identity we know that $\varepsilon_{FX} \circ F\eta_X = \id_{FX}$.
  Consider the naturality square of $\varepsilon$ applied to $F\eta_X$:
  \begin{center}
    \begin{tikzcd}
      FGFX \ar{r}{FGF\eta_X} \ar{d}{\varepsilon_{FX}} & FGFGFX \ar{d}{\varepsilon_{FGFX}} \\
      FX \ar{r}{F\eta_X} & FGFX
    \end{tikzcd}
  \end{center}
  Combining this with the assumption $GF\eta_X = \eta_{GFX}$, we get that $F\eta_X \circ \varepsilon_{FX} = \varepsilon_{FGFX} \circ FGF\eta_X = \varepsilon_{FGFX} \circ F\eta_{GFX} = \id_{FGFX}$, where the last equality is given by the triangle identity.
  Thus, $F\eta_X$ is an isomorphism, with inverse $\varepsilon_{FX}$.

  Now assume $F\eta_X$ is an isomorphism.
  By functoriality, $GF\eta_X$ is also an isomorphism with inverse $\mu_X$.
  Since $\eta_{GFX}$ also has $\mu_X$ as inverse, we get that $GF\eta_X = \eta_{GFX}$ by uniqueness of inverses.
\end{proof}

Let us apply the above discussion to the contravariant functor $(\argmhole)^\ast$.
This functor is right adjoint to itself, since $\Hom_R(A, B^\ast) \simeq \Hom_R(B,A^\ast)$.
To fit this into the above discussion, we take $F = (\argmhole)^\ast : \modcat{R} \to \modcat{R}^{\op}$
and $G = (\argmhole)^\ast : \modcat{R}^{\op} \to \modcat{R}$.
The unit and counit of this adjunction are both given by the evaluation map $\ev_A : A \to A^{\ast\ast}$.
Thus, the associated monad is $T \coloneq (\argmhole)^{\ast\ast}$, with multiplication $\mu_A \coloneq (\ev_{A^\ast})^\ast$.
A module $X$ is $T$-reflexive if $\ev_X$ is an isomorphism, which is equivalent to $X$ being reflexive.
We say that a module $X$ is \notion{torsionless} if it is $T$-torsionless for this $T$,
i.e., if $\ev_X$ is a monomorphism.
This is equivalent to being torsionless in the sense of H. Bass~\cite{bass1960} and motivates the name.
We can specialize \Cref{prp:monad-well-pointedness-torsionless-epi} with the improvement from \Cref{prp:monad-adjunction-well-pointed-improvement} to this situation.

\begin{corollary}\label{cor:dbl-dual-reflexive-iss-dual-reflexive}
  Let $A$ be an $R$-module. The following are equivalent:
  \begin{enumerate}[noitemsep]
    \item $A^{\ast\ast}$ is reflexive
    \item $A^\ast$ is reflexive
    \item $\ev_A$ is torsionless-epic. \earlyqed
  \end{enumerate}
\end{corollary}

\begin{lemma}\label{double-dual-tensor-finitely-adequate}
Let $A$ and $B$ be finitely adequate modules. Then $A \dbldualtensor B$ is finitely adequate.
\end{lemma}

\begin{proof}
By \Cref{prp:fin-adeq-hom-closure} we have that $(A \otimes B)^\ast = \Hom_R(A,\Hom_R(B,R))$ is finitely adequate, and therefore so is $A \dbldualtensor B$.
\end{proof}

\begin{proposition}\label{prp:dbl-dual-tensor-factorization}
  Let $A$, $B$, and $C$ be finitely adequate modules. Then every map $f : A \otimes B \to C$ factors uniquely through $\ev_{A \otimes B} : A \otimes B \to A \dbldualtensor B$.
\end{proposition}

\begin{proof}
  By \Cref{prp:monad-factorization}, we see that $f$ factors uniquely through $\ev_{A \otimes B}$ if both $A \dbldualtensor B$ and $C$ are reflexive. But $A \dbldualtensor B$ is finitely adequate by \Cref{double-dual-tensor-finitely-adequate} and $C$ is assumed finitely adequate, so we can conclude by \Cref{prp:can-dual-iso}.
%  The module $C$ is reflexive by \Cref{thm:can-dual-iso}, as it is finitely adequate.
%  The double dual tensor product $A \dbldualtensor B$ is reflexive if $(A \otimes B)^\ast$ is reflexive by \Cref{cor:dbl-dual-reflexive-iss-dual-reflexive}.
%  Now, $(A \otimes B)^\ast = \Hom_R(A \otimes B, R) \simeq \Hom_R(A, B^\ast)$, which is reflexive by \Cref{prp:fin-adeq-hom-closure} since $A$ and $B$ are finitely adequate.
\end{proof}

\begin{corollary}\label{cor:dbl-dual-tensor-adjunction}
  Let $M$ be a finitely adequate module.
  The functor $(\argmhole) \dbldualtensor M : \finadeq \to \finadeq$ is left adjoint to $\Hom_R(M,\argmhole) : \finadeq \to \finadeq$. \earlyqed
\end{corollary}

We finish this section by proving a general lemma of synthetic scheme theory, which is interesting in its own right and which we will need eventually to show that finitely adequate modules are not closed under tensor products.

For this, recall that a \emph{bundle} of $R$-modules on a scheme $X$ is simply a map $\mc F : X \to \Mod R$.
The \emph{global sections} of $\mc F$ are defined as $\Gamma(X,\mc F) :\equiv \prod_{x:X}\mc F_x$ and inherit the structure of an $R^X$-module by pointwise addition and multiplication.
In the current development of synthetic algebraic geometry, this notion takes on the role of $\mc O_X$-modules from classical algebraic geometry \cite[Chapter 7]{draft}.

\begin{lemma}
  \label{tensor-gamma-commute-on-affines}
  Let $X$ be an affine scheme and suppose that $\mathcal F, \mathcal G : X \to \Mod R$ are pointwise weakly quasi-coherent bundles.
  Then the map
  \[
    \vartheta_X : \Gamma(X,\mathcal F) \otimes_{R^X}\Gamma(X,\mathcal G) \to \Gamma(X,\mathcal F \otimes \mathcal G), \;\;
    f \otimes g \mapsto (x \mapsto f(x) \otimes g(x))
  \]
  is an $R^X$-linear isomorphism.
\end{lemma}

For the proof we need the following vanishing criterion for tensors.

\begin{remark}
  \label{null-tensor-criterion}
  Let $M$ and $N$ be $A$-modules over an arbitrary commutative ring $A$.
  For a tensor $k = \sum_{j=1}^n x_j \otimes y_j : M \otimes_A N$ we have $k = 0 : M \otimes_A N$ if and only if there merely are $p \geq n$, $q \geq 0$ together with elements $x_{n+1},\hdots,x_p : M$, $z_1,\hdots,z_q : N$ and a matrix $G : A^{p \times q}$, such that with
  \begin{align*}
    S &\equiv (x_1 \ \cdots \ x_p) : M^{1 \times p}, \\
    T &\equiv (y_1 \ \cdots \ y_n \; 0 \ \cdots \ 0)^T : N^{p \times 1}, \\
    U &\equiv (z_1 \ \cdots \ z_q)^T : N^{q \times 1}
  \end{align*}
  we have $SG = 0 : M^{1 \times q}$ and $GU = T : N^{p \times 1}$.
\end{remark}

For finitely generated $M$, this is the null-tensor lemma \cite[Lemma IV-4.14]{lombardi-quitte}.
The arbitrary case follows from the criterion that $k=0 : M\otimes_A N$ if and only if, merely, there is a finitely generated submodule $M'\subseteq M$ containing $x_1,\ldots,x_n$ such that $k = 0 : M' \otimes_A N$; see also \cite[Exercise IV-4]{lombardi-quitte}.
\begin{proof}
  Say $X = \Spec A$.
  We show injectivity of $\vartheta_X$ first.
  Let us first explain how this is a local problem.
  From commutative algebra we know that it is enough to find $a_1,\hdots,a_m:A$ with $(a_1,\hdots,a_m) = A$ such that $\vartheta_X[a_i^{-1}]$ is injective for all $i=1,\hdots,m$.
  As $\mc F$ and $\mc G$ are weakly quasi-coherent, we use \cite[Lemma 7.1.8]{draft} to see that the left vertical map in the commutative square
  \[ \begin{tikzcd}[column sep=large]
    \Gamma(X,\mc F)[a_i^{-1}] \otimes_{A[a_i^{-1}]} \Gamma(X,\mc G)[a_i^{-1}] \ar[d, "\simeq"]\ar[r, "\vartheta_X{[a_i^{-1}]}"]
    &
    \Gamma(X, \mc F \otimes \mc G)[a_i^{-1}] \ar[d]
    \\
    \Gamma(D(a_i),\mc F) \otimes_{R^{D(a_i)}}\Gamma(D(a_i),\mc G) \ar[r, "\vartheta_{D(a_i)}"]
    &
    \Gamma(D(a_i),\mc F \otimes \mc G)
  \end{tikzcd}
  \]
  is an isomorphism.
  Hence, the injectivity of all $\vartheta_{D(a_i)}$ implies the injectivity of $\vartheta_X$.
  Together with the fact that $(a_1,\hdots,a_m) = A$ if and only if $X = D(a_1) \cup \hdots \cup D(a_m)$, this means in particular that, for a given element of the kernel of $\vartheta_X$, it is enough to prove its vanishing after replacing $X$ by the members of a finite affine cover. We shall use this when invoking Zariski-local choice below.

  Now let $k:\Gamma(X,\mc F) \otimes_A \Gamma(X,\mc G)$ be such an element, i.e.\ $\vartheta_X(k)=0$.
  Merely, we may find $f_1,\hdots,f_n : \Gamma(X,\mc F)$ and $g_1,\hdots,g_n : \Gamma(X,\mc G)$ such that $k = \sum_{j=1}^n f_j \otimes_A g_j$ and $\sum_{j=1}^n f_j(x) \otimes g_j(x) = 0  : \mc F_x \otimes \mc G_x$ for all $x:X$.
  By virtue of \Cref{null-tensor-criterion}, we merely get the datum
  \[
    p_x \geq n, \;\;
    q_x \geq 0, \;\;
    S_x : \mc F_x^{1 \times p_x}, \;\;
    T_x : \mc G_x^{p_x \times 1}, \;\;
    U_x : \mc G_x^{q_x \times 1}, \;\;
    G_x : R^{p_x \times q_x},
  \]
  such that $S_xG_x = 0$ and $G_xU_x = T_x$. Here, $S_x$, $T_x$ and $U_x$ are defined as in the remark.
  Explicitly, the first $n$ entries of $S_x$ are $f_1(x), \hdots, f_n(x)$  and the remaining are supplied datum by the remark.
  The first $n$ entries of $T_x$ are $g_1(x), \hdots, g_n(x)$ while the trailing entries are zero.
  Using Zariski-local choice while shrinking $X$ if necessary as explained in the first paragraph, we get functions $p,q : X \to \mathbb N$ such that $p_x \geq n$ for all $x:X$, together with dependent functions $S : (x:X) \to \mc F_x^{1 \times p_x}$, $T : (x:X) \to \mc G_x^{p_x \times 1}$, $U : (x:X) \to \mc G_x^{q_x \times 1}$ and $G : (x:X) \to R^{p_x \times q_x}$ such that $SG = 0$ and $GU = T$.
  The functions $p$ and $q$ are bounded by \cite[Theorem 3.3.3]{draft}.
  Furthermore, because equality in $\mathbb N$ is decidable and hence open, all fibers of $p$ and $q$ are open subschemes of $X$.
  It follows that $(\fib_p(n))_{n:\mathbb N}$ and $(\fib_q(n))_{n:\mathbb N}$ are both finite open covers of $X$.
  Again shrinking $X$, we thus may assume that $p$ and $q$ are constant.
  Therefore, we have $S : \Gamma(X,\mc F)^{1 \times p}$, $T : \Gamma(X,\mc G)^{p \times 1}$, $U : \Gamma(X,\mc G)^{q \times 1}$ and $G : \Gamma(X, R)^{p \times q} \simeq A^{p \times q}$.

  By construction, the first $n$ entries of $S$ and $T$ are given by $f_1, \hdots, f_n$ and $g_1, \hdots, g_n$ respectively and the trailing entries of $T$ are zero.
  Therefore, the reverse implication of \Cref{null-tensor-criterion} gives that $k = \sum_{j=1}^n f_j \otimes g_j = 0$\footnote{
    Alternatively, it is perhaps enlightening to do this calculation by foot:
    \[
      \textstyle
      \sum_{j=1}^n f_j \otimes g_j = \sum_{i=1}^p S_i \otimes T_i = \sum_{i,k} S_i \otimes G_{ik}U_k = \sum_k(\sum_i S_iG_{ik}) \otimes U_k = 0 \rlap{.}
    \]
    This is essentially the proof of the reverse implication of the remark.
  }.

  Lastly we verify surjectivity.
  Let $s:\Gamma(X,\mc F \otimes \mc G)$ and set $H :\equiv \Gamma(X,\mc F) \otimes_A \Gamma(X,\mc G)$.
  We know that $s(x) : \mc F_x \otimes \mc G_x$ merely is a finite sum of elementary tensors.
  Applying Zariski-local choice with boundedness again, we find an affine cover $D(a_1),\dots,D(a_n) \subseteq X$ together with sections $h_i : \Gamma(D(a_i), \mc F) \otimes_{R^{D(a_i)}} \Gamma(D(a_i),\mc G)$ such that $s|_{D(a_i)} = \vartheta_{D(a_i)}(h_i)$.
  As above, let us identify $H[a^{-1}] \simeq \Gamma(D(a),\mc F) \otimes_{A[a^{-1}]} \Gamma(D(a), \mc G)$ for all $a:A$.
  Then we regard $h_i$ as an element of $H[a_i^{-1}]$ and  the map $\vartheta_{D(a_ia_j)}$ is of type $H[(a_ia_j)^{-1}] \to \Gamma(D(a_ia_j),\mc F \otimes \mc G)$.
  We have just verified that it is injective, thus the images of $h_i$ and $h_j$ in $H[(a_ia_j)^{-1}]$ agree and glue to an element $h : H$ with $\vartheta_X(h)|_{D(a_i)} = \vartheta_{D(a_i)}(h_i) = s|_{D(a_i)}$ for all $i$.
  Therefore, also $\vartheta_X(h) = s$.
\end{proof}

\begin{corollary}
  We have the following.
  \begin{enumerate}
  \item[i)]
    \label{wqc-closed-under-tensorprod}
    If $M$ and $N$ are (weakly) quasi-coherent $R$-modules, then so is $M \otimes N$.

  \item[ii)]
    \label{wqc-closed-under-dep-prod}
    If $\mc F : X \to \Mod R$ is a bundle of (weakly) quasi-coherent $R$-modules on an affine scheme, then $\Gamma(X,\mc F)$ is (weakly) quasi-coherent.
  \end{enumerate}
\end{corollary}

\begin{proof}
  The first item is clear.
  For the second one let $B$ be an arbitrary finitely presented $R$-algebra and suppose $X = \Spec A$.
  Note that $B$ is quasi-coherent as an $R$-module.
  In the following we conflate notation for $B$ and its induced constant bundle $x \mapsto B$ on $X$.
  Consider the composition of isomorphisms:
  \begin{align*}
    \Gamma(X, \mc F) \otimes_R B &\overset{\simeq}{\longrightarrow} \Gamma(X,\mc F) \otimes_A (A \otimes_R B) \\
                                 &\overset{\simeq}{\longrightarrow} \Gamma(X,\mc F) \otimes_A \Gamma(X,B) \\
                                 &\overset{\simeq}{\longrightarrow} \Gamma(X, \mc F \otimes B) \\
                                 &\overset{\simeq}{\longrightarrow} \Gamma(X, {\mc F}^{\Spec B}) \\
                                 &\overset{\simeq}{\longrightarrow} \Gamma(X,\mc F)^{\Spec B}
  \end{align*}
  We omit the verification that this composition is the given map in the definition of (weak) quasi-coherence.
  This shows the claim for quasi-coherence and, by restricting $B$ to algebras of the form $R[r^{-1}]$, also for weak quasi-coherence.
\end{proof}
