\begin{proposition}[Effective Projective Nullstellensatz]
  \label{effective-nullstellensatz}
  Let $F_1 : H^0(\bP^n,\mc O(e_1)), \hdots, F_m : H^0(\bP^n,\mc O(e_m))$ with $e_1,\hdots,e_m : \bZ$.
  We fix an arbitrary $N : \bN$, such that $N - \sum_{j:J} e_j \geq -n$, where $J$ ranges over all decideable subsets of $\{1, \hdots, m\}$.
  Set $\mc E :\equiv \mc O(-e_1) \oplus \hdots \oplus \mc O(-e_m)$. Then the following are equivalent.
  \begin{enumerate}
    \item[i)] The sections $F_1,\hdots,F_m$ have no common zero, i.e.\ $D_+(F_1,\hdots,F_m) = \bP^n$.
    \item[ii)] We have $ (x_0,\hdots,x_n)^N \subseteq (F_1,\hdots,F_m)$.
    \item[iii)] The $R$-linear map
      \[
        A : H^0(\bP^n,\mc E(N)) \longrightarrow H^0(\bP^n,\mc O(N)), \;
        (G_1,\hdots,G_m) \mapsto \sum G_jF_j
      \]
      is surjective.
  \end{enumerate}
\end{proposition}

Before we prove this result, let us recall a useful classical lemma.
Note, that it does not use any axioms of synthetic algebraic geometry and would work in plain HoTT over any base ring.

\begin{lemma}
  \label{exactness-criterion-Koszul-complex}
  Let $M$ be an $R$-module and $f : M \to R$ an $R$-linear map.
  Then the associated Koszul complex
  \[
    \textstyle
    K_\bullet(f) \;\;:\;\;
    \cdots          \longrightarrow
    \bigwedge^mM    \overset{d_m}{\longrightarrow}
    \cdots          \overset{d_3}{\longrightarrow}
    \bigwedge^2 M   \overset{d_2}{\longrightarrow}
    M               \overset{f}{\longrightarrow}
    R               \longrightarrow 0 \rlap{.}
  \]
  is exact if and only if $f$ is surjectve.
\end{lemma}

\begin{proof}
  If $K_\bullet(f)$ is exact, then $f$ is clearly surjective.
  Conversely, assume $f$ is surjective.
  For a unified notation set $d_1 :\equiv f$ and $d_0 :\equiv 0$.
  Merely choose a preimage $m:M$ of $1:R$.
  Since exactness is a proposition we may eleminate this truncation.
  Set
  \[
    \textstyle
    h_k : \bigwedge^kM \to \bigwedge^{k+1}M, \;
    \omega \mapsto m \wedge \omega \rlap{.}
  \]
  Then we have a chain homotopy $d_{k+1}h_k + h_{k-1}d_k = id$ and therefore $K_\bullet(f)$ is contractible and thus exact.
\end{proof}

\begin{proof}[Proof of \Cref{effective-nullstellensatz}]
  The implication $iii) \to ii)$ is clear.
  For the implication $ii) \to i)$ suppose $p : \bP^n$ is arbitrary. Without loss of generality assume $p \in D_+(x_0)$.
  We can find a relation $x_0^N = \sum G_jF_j : H^0(\bP^n,\mc O(N))$, implying that $0 \neq p_0^N = \sum G_j(p)F_j(p)$.
  By locality of $R$ we obtain $F_1(p) \neq 0\lor \hdots \lor F_m(p) \neq 0$ as desired.
  Lastly, we show $i) \to iii)$.
  For this we set $U :\equiv D_+(F_1,\hdots,F_m)$ and assume that for all $p:\bP^n$ we have $U(p)$.
  Furthermore, we define a map
  \[
    \varepsilon : \mc E \longrightarrow \mc O, \;
    (G_1,\hdots,G_m) \mapsto \sum G_jF_j \rlap{.}
  \]
  of bundles on $\bP^n$.
  The induced $R$-linear map $H^0(\bP^n,\varepsilon(N)) : H^0(\bP^n,\mc E(N)) \to H^0(\bP^n,\mc O(N))$ of finite free $R$-modules is precisely $A$.
  Consider the Koszul complex
  \[
    \textstyle
    K_\bullet(\varepsilon) \;\;:\;\;
    \cdots \longrightarrow
    0 = \bigwedge^{m+1}\mc E \overset{d_{m+1}}{\longrightarrow}
    \bigwedge^m\mc E         \overset{d_m}{\longrightarrow}
    \cdots                   \overset{d_3}{\longrightarrow}
    \bigwedge^2 \mc E        \overset{d_2}{\longrightarrow}
    \mc E                    \overset{\varepsilon}{\longrightarrow}
    \mc O                    \longrightarrow 0 \rlap{.}
  \]
  More precisely, this is pointwise the Koszul complex of $\varepsilon_p : \mc E_p \to R$, i.e.\ $K_\bullet(\varepsilon)_p \equiv K_\bullet(\varepsilon_p)$.
  Since we assumed $U(p)$, the map $\varepsilon_p : \mc E_p \to R$ is surjective.
  Then according to \Cref{exactness-criterion-Koszul-complex}, $K_\bullet(\varepsilon_p)$ is exact.
  Hence $K_\bullet(\varepsilon)$ is pointwise exact.
  Splicing $K_\bullet(\varepsilon)$, we obtain short exact sequences
  \[
    \textstyle
    0 \longrightarrow \im(d_{k+1}) \longrightarrow \bigwedge^k\mc E \longrightarrow \im(d_k) \longrightarrow 0
  \]
  for $k \geq 0$.
  Since $\bigwedge^k\mc E(N) = \bigoplus_{i_1 < \hdots < i_k} \mc O(N-e_{i_1} - \hdots - e_{i_k})$ and $N - e_{i_1} - \hdots - e_{i_k} \geq -n$, we conclude that $H^q(\bP^n,\bigwedge^k\mc E(N))=0$ for $q > 0$.
  Therefore, after twisting the above sequences by $\mc O(N)$, the associated long exact sequences yield a chain of isomophisms
  \begin{align*}
    H^1(\bP^n,\im(d_2(N))) &= H^2(\bP^n,\im(d_3(N))) \\
                           &= \hdots \\
                           &= H^m(\bP^n,\im(d_{m+1}(N))) \\
                           &= 0 \rlap{.}
  \end{align*}
  The last isomorphism follows since $d_{m+1}=0$.
  Again using the long exact sequence of cohomology for $k=1$, we get an exact excerpt $H^0(\bP^n,\mc E(N)) \to H^0(\bP^n, \mc O(N)) \to H^1(\bP^n,\im(d_2(N)))=0$, showing that the first map, which is $A$, is surjective.
\end{proof}

\begin{corollary}
  \label{P^n-compact}
  The projective spaces $\bP^n$ are compact for all $n \geq 0$.
\end{corollary}

\begin{proof}
  Let $U : \bP^n \to \Open$ be an open predicate on $\bP^n$.
  We need to show that $\prod_{p:\bP^n}U(p)$ is open.
  Merely, we can find homogeneous polynomials $F_1,\hdots,F_m$ such that $U = D_+(F_1,\hdots,F_m)$.
  Using \Cref{effective-nullstellensatz}, we thus see that $\prod_{p:\bP^n}U(p)$ is equivalent to the surjectivity of a matrix $A$.
  Since the surjectivity of a matrix is an open condition, we are done.
\end{proof}

\begin{proposition}[Weak Bertini]

This version of Bertini and the preceding remark allows us to employ a very classical proof of the following existence result for finite maps.

\begin{corollary}
  \label{etale-merely-finite-map}
  Let $X$ be a smooth scheme of dimension $n$ and suppose $\mc L$ is a very ample bundle on $X$.
  Then the étale-merely is a finite map $f : X \to \bP^n$ such that $\mc L = f^\ast\mc O(1)$.
\end{corollary}

\begin{proof}
  Suppose $\mc L$ embeds $X$ via $i : X \hookrightarrow \bP^N$, i.e.\ $i^\ast\mc O(1) = \mc L$.
  According to \Cref{weak-bertini} we can iteratively find hyperplanes $H_0, \hdots, H_n \subseteq \bP^N$ such that $X \cap H_0 \cap \hdots \cap H_n$ is smooth of dimension $-1$.
  Then we automatically have
  \[
    X \cap H_0 \cap \hdots \cap H_d = \emptyset \rlap{.}
  \]
  If $l_0, \hdots, l_n$ are linear homogeneous polynomials cutting out $H_0, \hdots, H_n$, then we have just verified that they have no common zero on $X$.
  Hence the universal property of $\bP^n$ yields a map $f :\equiv [l_0 : \hdots : l_n] : X \to \bP^n$ such that $f^\ast\mc O(1) = \mc L$.
\end{proof}

In \Cref{merely-finite-map} we will invest significant effort to upgrade this result to mere existence.
In that case however, we cannot use linear sections in $\mc O(1)$ to construct the map $f$ and instead have to resort to sections of some $\mc O(d)$.
It is however noteworthy, that even this weaker existence result can be quite useful, since a lot of interesting results turn out to be étale sheaves anyway.
The type of dualizing bundles on a scheme is an instance of such a proposition that is an étale sheaf.

  \[
    Z_t :\equiv X \cap H_{h_0(t_0)} \cap ... \cap H_{h_j(t_j)}

  For a smooth closed subscheme $Z$ of $\bP^N$ let us denote by $U_Z \subseteq \bP^N$ the non-empty open subset from \Cref{weak-bertini}.

    V_{\leq j} :\equiv \bigcap_{t:T_{\leq j}}U_{Z_t}
                \equiv \sum_{p : \bP^N} \prod_{t:T_{\leq j}} p \in U_{Z_t} \rlap{.}

  We claim that $\neg\neg V_{\leq j}$.
  Indeed, we know that $\neg\neg \big( T_{\leq j} = \{1, \hdots, d_{\leq j}\} \big)$, where $d_{\leq j} :\equiv d_0 \cdot \hdots \cdot d_j$, so we may assume that $T_{\leq j}$ is a standard finite set.

A first useful consequence of this result is the following bound on the cohomological dimension.

\begin{corollary}
  If $X$ is a projective scheme that is smooth of dimension $n$ and $\mc F : X \to \Mod R$ is a weakly quasi-coherent bundle, then
  \[
    H^p(X,\mc F) = 0
  \]
  for all $p > n$.
\end{corollary}

\begin{proof}
  Using \Cref{merely-finite-map}, we may choose a finite map $f : X \to \bP^n$.
  Then the claim reduces to the corresponding one on $\bP^n$.
  Indeed, $f$ is affine and thus $H^p(X,\mc F) = H^p(\bP^n,f_\ast\mc F) = 0$ for $p > n$.
\end{proof}



\subsection{Cohomology of Projective Bundles}

\subsubsection{Finitely Presented Bundles}

We now come to study the structure of cohomology of finitely presented bundles on $\bP^n$.
As expected, the cohomology of such bundles is not usually finitely presented.
This already fails for the global sections of vector bundles on $\bP^1$, as the following example illustrates.

\begin{example}
  \label{cohomology-of-vectorbundle-prescribed-by-matrix}
  We work over $\bP^n$ for an arbitrary $n \geq 1$.
  Suppose $A : R^s \to R^t$ is a matrix.
  Consider the exact sequence
  \[
    0 \longrightarrow \mathcal O(-n-1)^{\oplus s}
    \overset{(A,x_0,\hdots, x_n)}{\longrightarrow} \mathcal O(-n-1)^{\oplus t} \oplus \mathcal O(-n)^{\oplus (n+1)s}
    \longrightarrow \mathcal F_A \longrightarrow 0,
  \]
  where $\mc F_A$ is the cokernel of the previous map.
  Since we multiply by all $x_0, \hdots, x_n$,  this first map is pointwise injective and therefore $\mc F_A$ is a vector bundle of rank $(n+1)s + t - s$.
  As $H^i(\bP^n, \mc O(-n)) = 0$ for all $0 \leq i \leq n$, $H^n(\bP^n,\mc O(-n-1)) = R$ and $H^n(\bP^n,\mc O(-n)) = 0$, the long exact sequence of cohomology gives
  \[
    H^i(\bP^n, \mc F_A) = \begin{cases}
      \coker(A) &\mbox{if } i = n \\
      \ker(A)  &\mbox{if } i = n-1 \\
      0        &\mbox{else}
    \end{cases}
    \rlap{.}
  \]
  In case of $n=1$, this implies that it is not true that $H^0(\bP^1,\mathcal F_A)$ is finitely presented, or that $H^1(\bP^1, \mc F_a)$ is finitely copresented, for all matrices $A$.
  Interestingly enough, this also shows that is not true that all vector bundles on $\bP^1$ split into line bundles, since then $H^0(\bP^1, \mathcal F_A)$ would need to be finite free for all $A$.
\end{example}

Now consider a weakly quasi-coherent and finitely generated bundle $\mathcal F$ on an affine scheme $\Spec A$.
It follows readily from an application of Zariski-local choice, that then $\Gamma (X,\mathcal F)$ is a finitely generated $A$-module as well.
Careful analysis of the proof of this result affords us the following strengthening.

\begin{proposition}
  \label{wqc+fg-bundle-is-explicitly-generated}
  Given an affine scheme $X=\Spec A$ and a bundle $\mathcal F : X \to \Mod{R}$ of $R$-modules the following holds.
  \begin{enumerate}
    \item[i)] If $\mathcal F$ is a bundle of weakly quasi-coherent and finitely generated modules, then there merely is a natural number $N : \mathbb N$ and an exact sequence
    \[
      \mathcal O_X^{\oplus N} \longrightarrow \mathcal F \longrightarrow 0.
    \]
    \item[ii)] If $\mathcal F$ is a bundle of finitely presented modules, then there merely are natural numbers $N,M : \mathbb N$ and an exact sequence
    \[
      \mathcal O_X^{\oplus M} \longrightarrow \mathcal O_X^{\oplus N} \longrightarrow \mathcal F \longrightarrow 0.
    \]
  \end{enumerate}
\end{proposition}

\begin{proof}
  We first show part $i)$.
  By virtue of Zariski-local choice and boundedness there merely is an open affine cover $D(f_1), ..., D(f_m) \subseteq X$ as well as surjective $R$-module morphisms $\alpha_{i,x} : R^n \to \mathcal F_x$ for all $i=1, ..., m$ and $x:D(f_i)$.
  Since $\mathcal F$ is weakly quasi-coherent, the induced $A_{f_i}$-linear map
  \[
    \textstyle\prod_{x:D(f_i)}\alpha_{i,x} :  A_{f_i}^n \longrightarrow \Gamma (D(f_i),\mathcal F) = \Gamma (X,\mathcal F)_{f_i}
  \]
  is again surjective. We may merely find $s_{i1}, ..., s_{in}:\Gamma (X,\mathcal F)$ and $n_{ij}:\mathbb{N}$ such that this map takes the standard basis vector $e_j$ to $s_{ij}/f_i^{n_{ij}}$.
  hen for all $x:X$ the $R$-linear maps
  \[
    \beta_x : R^{m \times n} \longrightarrow \mathcal F_x, \;
    e_{ij} \longmapsto s_{ij}(x)
  \]
  are surjective, hence we win.
  he second part follows directly from the first together with lemma \cite[Chapter IV, 1.0 Lemma]{lombardi-quitte}.
\end{proof}

\begin{proposition}
  \label{fp-module-bundle-globally-generated}
  Given a bundle $\mathcal F : \bP^n \to \Mod{R}$ of $R$-modules the following holds.
  \begin{enumerate}
    \item[i)] If $\mathcal F$ is a bundle of weakly quasi-coherent and finitely generated modules, then there merely is a natural number $m : \mathbb N$ and an exact sequence
    \[
      \mathcal O(q)^{\oplus m}\longrightarrow \mathcal F \longrightarrow 0 \rlap{.}
    \]
    \item[ii)] If $\mathcal F$ is a bundle of finitely presented modules, then there merely are natural numbers $m,l\geq 0$, $p ,q\leq 0$ and an exact sequence
    \[
      \mathcal O(p)^{\oplus l}     \longrightarrow
      \mathcal O(q)^{\oplus m}     \longrightarrow
      \mathcal F\longrightarrow 0  \rlap{.}
    \]
  \end{enumerate}
\end{proposition}

\begin{proof}
  Let $U_i = D(x_i)$ be the standard affine cover of $\bP^n$.
  By virtue of lemma \Cref{wqc+fg-bundle-is-explicitly-generated} we can find sections $s_{ij} : (x:U_i) \to \mathcal F_x$ such that $s_{i1}(x), ..., s_{im}(x)$ generate $\mathcal F_x$ for all $x:U_i$.
  Using lemma \Cref{extension-of-sections-by-twisting} we find $n:\mathbb{N}$ such that $x_i^ns_{ij}$ extends to a global section $t_{ij}:\Gamma (\bP^n,\mathcal F(n))$.
  hen the sections $\{t_{ij}\}_{ij}$ generate $\mathcal F(n)$ at every point $x:\bP^n$.

  For the second part we again use lemma \cite[Chapter IV, 1.0 Lemma]{lombardi-quitte} to see that the kernel of a surjection from a finite free bundle to a finitely presented bundle is finitely generated.
  We then apply the first part to this kernel to deduce the second part.
\end{proof}

\begin{corollary}\label{top-cohomology-of-fp-bundle-is-fp}
  Let $\mathcal F: \bP^n \to \Mod R$ be a bundle of $R$-modules.
  If $\mathcal F$ is pointwise finitely presented, then $H^n(\bP^n, \mathcal F)$ is a finitely presented $R$-module as well.
  Dually, if $\mathcal F$ is pointwise finitely copresented, then $H^0(\bP^n, \mathcal F)$ is a finitely copresented $R$-module.
\end{corollary}

\begin{proof}
  For the first part choose an exact sequence $\mathcal O(p)^{\oplus l}\to \mathcal O(q)^{\oplus m}\to \mathcal F\to 0$ and let $\mathcal K$ be the kernel of the surjection $\mathcal O(q)^{\oplus m}\to \mathcal F$.
  he long exact sequence of cohomology gives an exact sequence
  \[
    ... \to H^n(\bP^n,\mathcal K) \longrightarrow
    H^n(\bP^n, \mathcal O(q)^{\oplus m}) \longrightarrow
    H^n(\bP^n,\mathcal F) \to 0.
  \]
  he module in the middle is clearly finite free and the module to the left is finitely generated,  since the canonical surjection $\mathcal O(p)^{\oplus l} \to \mathcal K$ induces a surjection $H^n(\bP^n, \mathcal O(p)^{\oplus l}) \to H^n(\bP^n, \mathcal K)$.
  herefore $H^n(\bP^n, \mathcal F)$ is finitely presented.

  For the second part note that since dualizing is an involution, we may find an exact sequence $0 \to \mathcal F \to \mathcal O(-q)^{\oplus m} \to \mathcal O(-p)^{\oplus l}$. We then finish the proof by arguing dually to the first part.
\end{proof}

\begin{lemma}
If $M$ is a finitely adequate module then the diagonal
\[
  \Delta : M \longrightarrow M^{\bP^n}
\]
is an equivalence for all $n \geq 0$.
\end{lemma}

\begin{proof}
  Let $p : \bP^n$ be any point and $\varepsilon : M^{\bP^n} \to M$ the evaluation at $p$.
  Then clearly $\varepsilon \circ \Delta = id$.
  To see that $\Delta \circ \varepsilon = id$, let $f : \bP^n \to M$ be arbitrary.
  Then for any $\lambda : M^\ast$ we have that $\lambda \circ f : \bP^n \to R$ is the constant value $\lambda (f(p))$.
  But then for all $x:\bP^n$ we see that $f(x) - f(p)$ maps to zero under the canonical map $M \to M^{\ast\ast}$.
  As this map is injective, we obtain $f(x) - f(p) = 0$.
\end{proof}

Observe that in the proof we did not use the full hypothesis of $M \to M^{\ast\ast}$ being an equivalence, but only its injectivity.
Therefore the statement is more generally true for $R$-modules $M$ with this weaker property.

We now apply the results of this section to prove some special cases in which projective schemes are closed under $\sum$-types.
In the following we understand $\bP(M)$ as the set of rank $1$ quotients of an $R$-module $M$.

\begin{lemma}
  \label{projectivization-is-projective}
  Let $X$ be a projective scheme.
  If $\mc F: X \to \Mod R$ is a pointwise finitely presented bundle, then its projectivization
  \[
    \bP(\mc F) \equiv \sum_{x:X}\bP(\mc F_x)
  \]
  is a projective scheme.
\end{lemma}

\begin{proof}
  If $i:X \to \bP^n$ is any choice of a closed immersion, then $i_\ast\mc F$ is a finitely presented bundle and the canonical equivalence $i^\ast i_\ast\mc F \to \mc F$ induces another equivalence $\bP(i_\ast\mc F) \to \bP(\mc F)$.
  Thus we may assume $X = \bP^n$.
  Using \Cref{fp-module-bundle-globally-generated} we get an exact sequence $\mc E_1 \to \mc E_0 \to \mc F \to 0$ with $\mc E_0,\mc E_1$ finite direct sums of twisting bundles, say $\mc E_0 = \mc O(d)^{\oplus m}$.
Then $\mc E_{0,x} \to \mc F_x$ induces a closed immersion\footnote{
    If $A:R^n \to R^m$ is a matrix with cokernel $f:R^m \twoheadrightarrow F$, then we have an equivalence
    \[
      f^\ast:\bP(F) \to \{ (L,q):\bP(R^m) \;|\; q \circ A = 0 \}
      \rlap{,}
    \]
    where $q \circ A = 0$ is a closed proposition.
  }
  $\bP(\mc F_x) \lhook\joinrel\to \bP(\mc E_{0,x})$.
  Since closed propositions are closed under dependent sums, we get a closed immersion $\bP(\mc F) \lhook\joinrel\to \bP(\mc E_0)$.
  We deduce that $\bP(\mc E_0) = \bP(\mc E_0 \otimes \mc O(-d)) = \bP(\mc O^{\oplus m}) = \bP^n \times \bP^m$ is projective and are done.
\end{proof}

\subsubsection{Vector Bundles}

If we restrict our attention from finitely presented bundles to vector bundles, we can improve some of the results from the previous section drastically.
Recall that a vector bundle $\mathcal E : X \to \Mod R$ is a bundle of pointwise finite-free $R$-modules.
If $X$ is connected --- meaning that $\Bool \to \Bool^X$ is an equivalence --- the vector spaces $\mathcal E_x$ all have the same dimension.
As we are working over a local ring, a lot of the classical theorems and their proofs still apply mostly unchanged.

\begin{proposition}
  \label{cohomology-of-twisted-vectorbundles-on-P^n}
  Given a vector bundle $\mathcal E : \bP^n \to \Mod R$ the following holds.
  \begin{enumerate}
    \item[i)] The $R$-modules $H^p(\bP^n,\mc E)$ are finitely adequate for all $p \geq 0$.
    \item[ii)] There merely is a $d_0 : \bZ$ such that for all $d \geq d_0$ and $p > 0$  we have
      \[
        H^p(\bP^n,\mathcal E(d)) = 0.
      \]
      and that the $R$-module $H^0(\bP^n, \mc E(d))$ is finite free.
  \end{enumerate}
\end{proposition}

\begin{proof}
  Choose a surjection $\mathcal O(e)^{\oplus m} \to \mathcal E$.
  At any point $x:\bP^n$ its kernel $\mathcal K$ merely fits in an exact sequence $0 \to \mathcal K_x \to R^m \to R^l \to 0$ and since $R^l$ is projective, $\mathcal K_x$ is a direct summand of $R^m$ and therefore projective as well.
  As $R$ is local, we thus see that $\mathcal K$ is itself a vector bundle.
  Now we can follow the classical proofs presented in \cite{Hartshorne}:

  For the first part we reason by descending induction over $p$.
  For $p \geq n$ we already know the claim by \Cref{top-cohomology-of-fp-bundle-is-fp} and if $p < n$ we get an exact excerpt:
  \[
    \dots \to H^p(\bP^n,\mathcal O(e)^{\oplus m}) \to H^p(\bP^n, \mathcal E) \to H^{p+1}(\bP^n, \mathcal K) \to H^{p+1}(\bP^n,\mathcal O(e)^{\oplus m}) \to\dotsi\rlap{.}
  \]
  Note that the first module is zero and hence $H^p(\bP^n,\mathcal E)$ is the kernel of $H^{p+1}(\bP^n, \mathcal K) \to H^{p+1}(\bP^n,\mathcal O(e)^{\oplus m})$.
  These two modules are finitely adequate by induction hypothesis, hence so is the kernel.

  For the second part, we once again argue by descending induction over $p$. Similar to the first part, we know the claim for $p>n$.
  Consider the long exact sequence
  \[
    \dots \to H^p(\bP^n,\mathcal O(e+d)^{\oplus m}) \to H^p(\bP^n, \mathcal E(d)) \to H^{p+1}(\bP^n, \mathcal K(d)) \to \dotsi\rlap{.}
  \]
  for $p > 0$.
  If $e+d>0$, then $H^p(\bP^n,\mathcal O(e+d)^{\oplus m}) = 0$ and by induction hypothesis also $H^{p+1}(\bP^n,\mathcal K(e+d)) = 0$ for $d \gg 0$.
  Hence also $H^p(\bP^n, \mathcal E(d)) = 0$ for $d \gg 0$.

  It remains to see that $H^0(\bP^n, \mc E(d))$ is finite free.
  According to \Cref{top-cohomology-of-fp-bundle-is-fp}, $H^0(\bP^n,\mathcal E(d))$ is finitely copresented.
  The start of the long exact sequence of cohomology above gives us an exact excerpt $H^0(\bP^n, \mathcal O(e+d))^{\oplus m} \to H^0(\bP^n, \mathcal E(d)) \to H^1(\bP^n, \mathcal K(d))$.
  We have just verified that $H^1(\bP^n, \mc K(d)) = 0$ for $d \gg 0$, hence $H^0(\bP^n, \mc E(d))$ is finitely generated for those $d$.
  However, any module that is both finitely copresented and finitely generated is finite free, hence so is $H^0(\bP^n, \mathcal E(d))$.
\end{proof}

\begin{remark}
  The twist in the statement of \Cref{cohomology-of-twisted-vectorbundles-on-P^n} is necessary.
  Indeed, it follows from \Cref{cohomology-of-vectorbundle-prescribed-by-matrix}, that $H^0(\bP^1, \mc E)$ is not finite free, as well as that $H^1(\bP^1, \mc E)$ does not vanish for all vector bundles $\mc E$, as one can reach all (co-)kernels of all matrices $R^{s \times t}$ by varying the vector bundle $\mc E$.
\end{remark}

Using our finite free map machinery, we can extend this result to all smooth projective schemes.

\begin{corollary}
  \label{cohomology-of-twisted-vectorbundles-on-sm-projective-schemes}
  Let $X$ be a smooth projective scheme and suppose $\mc E$ is a vector bundle and $\mc L$ is a very ample bundle on $X$.
  Then the following holds.
  \begin{enumerate}
    \item[i)] The $R$-modules $H^p(X, \mc E)$ are finitely adequate for all $p \geq 0$.
    \item[ii)] There merely is a $d_0:\bZ$ such that for all $d \geq d_0$ and $p > 0$ we have
      \[
        H^p(X, \mc E \otimes \mc L^{\otimes d}) = 0
      \]
      and such that $H^0(X, \mc E \otimes \mc L^{\otimes d})$ is finite free.
  \end{enumerate}
\end{corollary}

\begin{proof}
  Suppose $X$ is smooth of dimension $n$ and let $\mc L$ be an arbitrary very ample bundle on $X$.
  By \Cref{merely-finite-map} we merely obtain a finite map $f : X \to \bP^n$ such that $f^\ast \mc O(1) = \mc L^{\otimes m}$ for some $m > 0$.
  Then $f$ is clearly affine and furthermore preserves vector bundles by \Cref{finite-map-of-equidimensional-smooth-schemes-preserves-vectorbundles}.
  Thus $H^p(X, \mc E) = H^p(\bP^n, f_\ast \mc E)$ is finitely adequate by \Cref{cohomology-of-twisted-vectorbundles-on-P^n}.
  For the second item note that $\mc F_r :\equiv f_\ast(\mc E \otimes \mc L^{\otimes r})$ is also a vector bundle on $\bP^n$ for $0 \leq r < m$.
  Applying \Cref{cohomology-of-twisted-vectorbundles-on-P^n} to the finitely many bundles $\mc F_0, \hdots, \mc F_{m-1}$ merely yields integers $d'_0, \hdots, d'_{m-1} : \bZ$ such that for all $0 \leq r < m$ and $d' \geq d'_r$, $H^p(\bP^n, \mc F_r(d'))$ vanishes for $p>0$ and is finite free for $p=0$.
  The projection formula of \Cref{projection-formula-II} the gives $f_\ast(\mc E \otimes \mc L^{\otimes md'+r}) = \mc F_r \otimes \mc O(d')$.
  Set $d_0 :\equiv m \cdot \max\{d'_0, \hdots, d'_{n-1}\} : \bZ$.
  An integer $d \geq d_0$ can be written as $d = md' +r$ for $0 \leq r < m$ and $d' \geq \max\{d'_0, \hdots, d'_{m-1}\}$.
  Then we have that
  \[
    H^p(X,\mc E \otimes \mc L^{\otimes d}) = H^p(\bP^n, \mc F_r (d'))
  \]
  vanishes for $p > 0$ and is finite free for $p=0$.
\end{proof}

\begin{remark}
  \label{Koszul-complex-of-vector-bundle}
  Suppose that $\mc E$ is a vector bundle on $\bP^n$.
  Consider the Koszul complex
  \[
    0 \longrightarrow \mc O(-(n+1)d)
      \longrightarrow \mc O(-nd)^{\binom{n+1}{n}}
      \longrightarrow \cdots
      \longrightarrow \mc O(-d)^{\binom{n+1}{1}}
      \longrightarrow \mc O
      \longrightarrow 0
  \]
  induced by $x_0^d, \hdots, x_n^d$, which is exact by \Cref{exactness-criterion-Koszul-complex}.
  Here we have used the identifications $\bigwedge^k\mc O(-d)^{n+1} = \mc O(-kd)^{\binom{n+1}{k}}$.
  Tensoring with $\mc E((n+1)d)$, we obtain another pointwise exact sequence
  \[
    0 \longrightarrow \mc E
      \longrightarrow \mc E(d)^{\binom{n+1}{n}}
      \longrightarrow \cdots
      \longrightarrow \mc E(nd)^{\binom{n+1}{1}}
      \longrightarrow \mc E((n+1)d)
      \longrightarrow 0
      \rlap{.}
  \]
  Splicing this resolution into short exact sequences and using \Cref{cohomology-of-twisted-vectorbundles-on-P^n}, we see that the complex obtained by applying $H^0(\bP^n, -)$ computes the cohomology of $\mc E$ for $d \gg 0$.
  This yields another proof that the cohomology of vector bundles on $\bP^n$ is finitely adequate.
\end{remark}

Classically, one defines the Euler characterstic of a locally free (or even coherent) sheaf $\mc F$ on $\bP^n_k$ over a field $k$ as $\sum_j (-1)^j \dim_k H^j(\bP^n_k, \mc F)$.
We cannot directly mimic this definition, as the cohomology of even vector bundles need not be finite free in our setting.
However, the theory of the previous section still makes it possible to define this invariant via a slight detour.

\begin{lemma}
  \label{axiomatized-euler-characteristic-for-P^n}
  There is a unique function $\chi : \Vect{\bP^n} \to \bZ$ satisfying the following items.
  \begin{enumerate}
    \item[i)] If $0 \to \mc E' \to \mc E \to \mc E'' \to 0$ is pointwise exact, then $\chi(\mc E') + \chi(\mc E'') = \chi(\mc E)$.

    \item[ii)] If $\mc E$ is such that $H^0(\bP^n, \mc E)$ is finite free and $H^{>0}(\bP^n, \mc E) = 0$, then $\chi(\mc E) = \dim_R H^0(\bP^n, \mc E)$.
  \end{enumerate}
  Furthermore, if for a given vector bundle $\mc E$ there is an integer $d_0:\bZ$ such that for all $d \geq d_0$ the vector bundles $\mc E(d)$ satisfies the conditions of $ii)$, then
  \[
    \chi(\mc E) = \sum_{j=1}^{n+1} (-1)^{j+1} \binom{n+1}{j} \dim_R H^0(\bP^n,\mc E(jd)) \rlap{.}
  \]
  In particular, this description is independent of $d \geq d_0$.
\end{lemma}

\begin{proof}
  Suppose $\mc E$ is an arbitrary vector bundle and let $\chi$ be a function satisfying the above conditions.
  We show first that $\chi(\mc E)$ is uniquely prescribed.
  According to \Cref{cohomology-of-twisted-vectorbundles-on-P^n}, there merely is an integer $d_0$ such that for all $d \geq d_0$ the bundles $\mc E(d)$ satisfy the conditions of the second item $ii)$.
  As we are proving a proposition, we may eliminate this truncation.
  Now consider the pointwise exact complex
  \[
    0 \longrightarrow \mc E
      \longrightarrow \mc E(d)^{\binom{n+1}{n}}
      \longrightarrow \cdots
      \longrightarrow \mc E(nd)^{\binom{n+1}{1}}
      \longrightarrow \mc E((n+1)d)
      \longrightarrow 0
      \rlap{.}
  \]
  from \Cref{Koszul-complex-of-vector-bundle}.
Let us abbreviate $h^0(a) :\equiv \dim_R H^0(\bP^n,\mc E(a))$ for $a \gg 0$ such that $H^0(\bP^n, \mc E(a))$ is finite free.
  Splitting this sequence into short exact sequences (note that the added bundles are again pointwise finite free), and iteratively applying the first condition, we arrive at a telescope sum, which, after cancelling terms, yields the first isomorphism in the following:
  \begin{align*}
    \chi(\mc E) &= \sum_{j=1}^{n+1} (-1)^{j+1} \chi(\mc E(jd)^{\binom{n+1}{j}}) \\
                &= \sum_{j=1}^{n+1} (-1)^{j+1} \binom{n+1}{j} \chi(\mc E(jd)) \\
                &= \sum_{j=1}^{n+1} (-1)^{j+1} \binom{n+1}{j} h^0(jd)
    \rlap{.}
  \end{align*}
  For the penultimate equality, we use that $\chi$ is additive by virtue of the first condition.
  The last equality follows from the second condition.
  Thus we see that $\chi(\mc E)$ is unique.
  It follows by function extensionality, that $\chi$ is unique as well.

  To prove existence we temporarily write $\chi_d(\mc E)$ for the above expression.
  Suppose another $e \geq d_0$ is given.
  We need to show $\chi_d(\mc E) = \chi_e(\mc E)$.
  Replacing $\mc E$ with $\mc E(ke)$, $k \geq 1$, in \Cref{Koszul-complex-of-vector-bundle}, we see that $\sum_{j=0}^{n+1}(-1)^j\binom{n+1}{j}h^0(ke+jd) = 0$.
  Thus we get
  \begin{align*}
    \chi_e(\mc E) &= \sum_{k=1}^{n+1} (-1)^{k+1} \binom{n+1}{k} h^0(ke) \\
                  &= \sum_{k=1}^{n+1} \sum_{j=1}^{n+1}(-1)^{j+1}(-1)^{k+1} \binom{n+1}{k} \binom{n+1}{j} h^0(ke+jd) \rlap{.}
  \end{align*}
  Since this expression is invariant when exchanging $(d,j)$ and $(e,k)$, it must be equal to $\chi_d(\mc E)$.
  Thus setting $\chi(\mc E)$ to be $\chi_d(\mc E)$ is well-defined.
  We omit the verification that this $\chi$ satisfies the two required conditions.
\end{proof}
