In this section we work in plain HoTT and study non-abelian torsors over an arbitrary type $X$. Our goal is to describe non-abelian torsors over $X$ trivialised by a given cover in terms of cocycles for this cover, i.e.\ to show that for any type $X$ with for each $x:X$ an inhabited type $Y(x)$ and a group $G(x)$, we have that $H^1(X,Y,G) = \check{H}^1(X,Y,G)$.

\begin{lemma}\label{first-cech-has-section-vanish}
Assume given a type $X$ with for each $x:X$ an inhabited type $Y(x)$ and $G(x)$ a group $G(x)$. If we merely have $\Pi_{x:X}Y(x)$ then $\check{H}^1(X,Y,G)$ is a singleton.
\end{lemma}

\begin{proof}
  Assume given $t: \Pi_{x:X} Y(x)$ and $\delta:\Pi_{x:X}G(x)^{Y(x)^2}$ a cocycle, we need to show it is cohomologous to $0$. It is enough to consider $\gamma:\Pi_{x:X}G(x)^{Y(x)}$ defined by $\gamma_x(y) = \delta_x(t_x,y)$.
\end{proof}

We define a map $\Phi$ from $H^1(X,Y,G)$ to $\check{H}^1(X,Y,G)$.

\begin{lemma}\label{phi-well-defined}
Assume given a type $X$ with for each $x:X$ an inhabited type $Y(x)$ and a group $G(x)$. Given $\alpha:\Pi_{x:X}\B G(x)$ and $\beta:\Pi_{x:X}\alpha(x)^{Y(x)}$, we define $\phi(\alpha,\beta):\Pi_{x:X}G(x)^{Y(x)^2}$ by $\phi(\alpha,\beta)_x(y_0,y_1) = \beta_x(y_0)^{-1}\cdot\beta_x(y_1)$. Then we have that:

\begin{enumerate}[(i)]
\item $\phi(\alpha,\beta)$ is a cocycle.
\item Given another $\beta':\Pi_{x:X}\alpha(x)^{Y(x)}$, we have that $\phi(\alpha,\beta)$ and $\phi(\alpha,\beta')$ are cohomologous.
\item We have an induced map $\Phi : H^1(X,Y,G)\to \check{H}^1(X,Y,G)$ defined by $\Phi(\alpha) = \phi(\alpha,\beta)$ for any $\beta:\Pi_{x:X}\alpha(x)^{Y(x)}$.
\end{enumerate}
\end{lemma}

\begin{proof}
As follows:

\begin{enumerate}[(i)]
\item Follows from direct computation.
\item To show that $\phi(\alpha,\beta)$ and $\phi(\alpha,\beta')$ are cohomologous, we consider $\gamma_x(y) := \beta'_x(y)^{-1}\cdot\beta_x(y)$. Given $x:X$ and $y_0,y_1:Y(x)$ we indeed have:
\[\phi(\alpha,\beta')_x(y_0,y_1)\cdot\gamma_x(y_1) = (\beta'_x(y_0)^{-1}\cdot \beta'_x(y_1))\cdot (\beta'_x(y_1)^{-1}\cdot\beta_x(y_1)) = \beta'_x(y_0)^{-1}\cdot\beta_x(y_1)\]
\[\gamma_x(y_0)\cdot\phi(\alpha,\beta)_x(y_0,y_1) = (\beta'_x(y_0)^{-1}\cdot \beta_x(y_0))\cdot(\beta_x(y_0)^{-1}\cdot\beta_x(y_1)) = \beta'_x(y_0)^{-1}\cdot\beta_x(y_1)\]
\item By \Cref{CechConditionEquivalentStatement}, for any $\alpha$ under consideration there exists a $\beta$ as above. By (ii) the class of $\phi(\alpha,\beta)$ in $\check{H}^1(X,Y,G)$ does not depend on the choice of such a $\beta$.
\end{enumerate}
\end{proof}
\begin{definition}
Recall that given a group $G$, a $G$-pseudotorsor is a type with a transitive and free right action of $G$ on $X$, and a $G$-torsor is an inhabited $G$-pseudotorsor.
\end{definition}

Now we will define a map $\Psi$ from $\check{H}^1(X,Y,G)$ to $H^1(X,Y,G)$.

\begin{lemma}\label{psi-well-defined}
Assume given an inhabited type $Y$ and a group $G$, as well as $\delta:G^{Y^2}$ a cocycle. %such that for all $y_0,y_1,y_2:Y$ we have $\delta(y_0,y_1)\cdot\delta(y_1,y_2) = \delta(y_0,y_2)$. 
We define $\psi(\delta) = \{\gamma:G^Y\ |\ \forall y_0,y_1.\, \gamma(y_0)\cdot\gamma(y_1)^{-1} = \delta(y_0,y_1)\}$. Then we have that:

\begin{enumerate}[(i)] 
  \item\label{Hdef} We can define $H:\psi(\delta)^Y$ by $H(y) = \lambda y'.\, \delta(y',y)$.
\item The type $\psi(\delta)$ can be given the structure of a $G$-torsor.
\item Given $\delta$ and $\delta'$ cocycles with $\iota:G^Y$ such that for all $y_0,y_1:Y$ we have $\iota(y_0)\cdot\delta(y_0,y_1) = \delta'(y_0,y_1)\cdot\iota(y_1)$, we have $\psi(\delta)=\psi(\delta')$.
\item We have an induced map $\Psi : \check{H}^1(X,Y,G)\to H^1(X,Y,G)$ defined by $\Psi(\delta) = \lambda x.\, \psi(\delta_x)$.
\end{enumerate}
\end{lemma}

\begin{proof}
We prove this as follows:

\begin{enumerate}[(i)]
\item We just need to check that for all $\bar{y},y_0,y_1$ we indeed have $\delta(y_0,\bar{y})\cdot\delta(y_1,\bar{y})^{-1} = \delta(y_0,y_1)$, but this follows from $\delta$ being a cocycle.

\item 
  $\psi(\delta)$ is inhabited by (i) and the assumption that $Y$ is inhabited. 
  We define the action of $g:G$ on $\gamma:\psi(\delta)$ by $\gamma\cdot g = \lambda y.\, \gamma(y)\cdot g$.
Note this action is well-defined and free. We will show it is also transitive.

Assume $\gamma,\gamma':\psi(\delta)$, then we claim that $g=\gamma(y)^{-1}\cdot\gamma'(y)$ does not depend on $y$. This follows from the observation that for any $y_0, y_1:Y$, $\gamma(y_0)\cdot\gamma(y_1)^{-1} = \delta_x(y_0,y_1) = \gamma'(y_0)\cdot\gamma'(y_1)^{-1}$. Since $Y$ is inhabited this gives us an element $g:G$ such that $\gamma(y)\cdot g = \gamma'(y)$ for all $y:Y$, which gives transtivity of our action. 

\item Observe that the map $\psi(\delta)\to \psi(\delta')$ sending $\gamma$ to $\lambda y.\, \iota(y)\cdot\gamma(y)$ is an equivalence of torsors.

\item From (ii), we get that $\psi(\delta):\B G$. So given $\delta$ a cocycle we have that $\Psi(\delta) = \lambda x.\, \psi(\delta_x)$ indeed gives an element in $H^1(X,G)$. From (iii) we have that cohomologous cocycles give the same element. From (i) we see that we have $\Pi_{x:X}\psi(\delta_x)^{Y(x)}$, so by \Cref{CechConditionEquivalentStatement} we indeed get an element in $H^1(X,Y,G)$. 
\end{enumerate}
\end{proof}

Now we prove the two maps are inverse to each other.

\begin{lemma}\label{PhiCircPsi}
  For $\Phi$ and $\Psi$ as above, we have that $\Phi\circ \Psi$ is the identity on $\check H^1(X,Y,G)$. 
\end{lemma}

\begin{proof}
  It is enough to prove that given a cocycle $\delta$, for all $x:X$ with $y_0,y_1:Y(x)$ we have that $H_x(y_0)^{-1}\cdot H_x(y_1) = \delta_x(y_0,y_1)$ where $H_x : Y(x)\to \psi(\delta_x)$ is defined as in \ref{Hdef} of \Cref{psi-well-defined}. As $H_x(\bar{y}) = \lambda y.\delta_x(y,\bar{y})$ it is sufficient to check that $(\lambda y.\delta_x(y,y_0))\cdot\delta_x(y_0,y_1) = \lambda y.\delta_x(y,y_1)$ which is clear as the action is defined pointwise.
\end{proof}

\begin{lemma}\label{PSiCircPhi}
  For $\Phi$ and $\Psi$ as above, we have that $\Psi\circ \Phi$ is the identity on $H^1(X,Y,G)$. 
\end{lemma}

\begin{proof}
Let $\alpha:\Pi_{x:X}\B G(x)$ with $\beta:\Pi_{x:X}\alpha(x)^{Y(x)}$. We must show that for all $x:X$ we have 
$\alpha(x) = \psi(\phi(\alpha,\beta)_x)$,
where
\[\psi(\phi(\alpha,\beta)_x) = \{\gamma:G(x)^{Y(x)}\ |\ \forall y_0,y_1.\, \gamma(y_0)\cdot\gamma(y_1)^{-1} = \beta_x(y_0)^{-1}\cdot\beta_x(y_1)\}\]

We define $f:\alpha(x)\to \psi(\phi(\alpha,\beta)_x)$ by $f(p) = \lambda y.\, \beta_x(y)^{-1}\cdot p$. $f$ is well defined as for all $y_0,y_1:Y(x)$ we have that:
\[f(p,y_0)\cdot f(p,y_1)^{-1} = \beta_x(y_0)^{-1}\cdot p \cdot (\beta_x(y_1)^{-1}\cdot p)^{-1} = \beta_x(y_0)^{-1}\cdot\beta_x(y_1)\]
Also $f$ commutes with the group action. We shall show it is an equivalence to conclude. 

In the other direction, assume given $\gamma:\psi(\phi(\alpha,\beta)_x)$. Then we have that $\beta_x(y)\cdot\gamma(y):\alpha(x)$ does not depend on the choice of $y:Y(x)$ by the hypothesis on $\gamma$, and since $Y(x)$ is inhabited this defines an element $g(\gamma):\alpha(x)$.

Next we check that $g(f(p)) = p$, for any $p:\alpha(x)$. Since we want to prove a proposition we can assume $y:Y(x)$, then we need to prove that $\beta_x(y)\cdot (\beta_x(y)^{-1}\cdot p) = p$ which is clear. Finally we check that for any $\gamma:\psi(\phi(\alpha,\beta)_x)$, we have $f(g(\gamma)) = \gamma$. Using that $g(\gamma)$ can be computed using any $y:Y(x)$, it is sufficient to prove that $\beta_x(y)^{-1}\cdot \beta_x(y) \cdot \gamma(y) = \gamma(y)$.
\end{proof}

\begin{corollary}\label{non-abelian-chech-is-sheaf-generic}
Assume given a type $X$ with for each $x:X$ an inhabited type $Y(x)$ and a group $G(x)$. Then $H^1(X,Y,G) = \check{H}^1(X,Y,G)$.
\end{corollary}

\begin{corollary}\label{non-abelian-chech-is-sheaf-generic-ayclic-cover}
Assume given a type $X$ with for each $x:X$ an inhabited type $Y(x)$ and a group $G(x)$. If $H^1(\Sigma_{x:X}Y(x), G\circ\pi_X) = 0$, then $H^1(X,G) = \check{H}^1(X,Y,G)$.
\end{corollary}
