\label{abelian-higher-cech}
In this section, we give a condition under which the \v Cech cohomology and the cohomology agree. 
\begin{lemma}\label{ZeroCohomologyAgree}
  Let $X$ be a type with $Y(x)$ an inhabited type and $A(x)$ an abelian group both depending on $x:X$. Then there is a natural isomorphism $\check H^0(X,Y, A) = H^0(X, A)$. 
\end{lemma}
\begin{proof}
We need to prove that the type of maps:
\[\{\alpha:\Pi_{x:X}A(x)^{Y(x)}\ |\ \forall x,y_0,y_1.\, \alpha_x(y_0)=\alpha_x(y_1)\}\]
is naturally isomorphic to $\Pi_{x:X}A(x)$. This is immediate from $Y(x)$ being inhabited sets.
\end{proof}

\begin{lemma}\label{cech-coefficient-lemma}
Assume given a type $X$ with $Y(x)$ a type and $A(x)$ an abelian group both depending on $x:X$. Then $\check{H}^n(X,Y,\lambda x.\, A(x)^{Y(x)})=0$ for all $n\geq 1$.
\end{lemma}

\begin{proof}
Indeed assume given:
\[\alpha : \Pi_{x:X} Y(x)^{n+1}\to A(x)^{Y(x)}\]
such that $\delta(\alpha) = 0$, i.e. for all $x:X$ and $y_0,\hdots, y_{n+1},y:Y(x)$ we have that:
\[\Sigma_{i=0}^{n+1}(-1)^i\alpha(x,y_0,\hdots,\widehat{y_i},\hdots ,y_{n+1},y) = 0\]
Then we define:
\[\beta : \Pi_{x:X} Y(x)^{n}\to A(x)^{Y(x)}\]
\[\beta(x,y_0,\hdots,y_{n-1},y) = (-1)^n\alpha(x,y_0,\hdots,y_{n-1},y,y)\]
and then:
\[\delta(\beta)(x,y_0,\hdots,y_{n},y) = (-1)^n\Sigma_{i=0}^{n} (-1)^i \alpha(x,y_0,\hdots,\hat{y_i},\hdots,y_n,y,y) \]
\[= \alpha(x,y_0,\hdots,y_n,y)\]
so $\delta(\beta) = \alpha$.
\end{proof}

\begin{definition}
  Let $X$ be a type, with for each $x:X$ an abelian group $A(x)$. A \v Cech cover for $X,A$ is a family of inhabited dependent sets $S(x)$ depending on $x:X$ such that for all $k\geq1$ the canonical map $$K(\Pi_{x:X} A(x)^{(S_x)^k}, n)\to \Pi_{x:X}(K(A(x),n))^{(S_x)^k}$$ is an equivalence. 
\end{definition}
\begin{lemma}\label{TrivialHasVanishingCohomology}
  $S$ is a \v Cech cover for $X,A$ iff $H^n(X,\lambda x. A(x)^{S(x)^k}) = 0$ for all $n,k\geq1$. 
\end{lemma}
\begin{proof}
  This follows directly from \Cref{ev-eq-acyclic}, \Cref{ev-embedding} and \Cref{acyclic-ev-surjective}.
\end{proof}

\begin{corollary}\label{CechExpCohomologiesAgree}
Assume given a type $X$ and for each $x:X$ an Abelian group $A(x)$, with a \v Cech cover $S$. Then for all $n:\N$, we have a natural isomorphism
$H^{n}(X,\lambda x.A(x)^{S(x)}) = 
\check H^{n}(X,S,\lambda x.A(x)^{S(x)})
$.
\end{corollary}

\begin{lemma}\label{inductive-definition-cohomology-general}
Assume given a type $X$ and for each $x:X$ an Abelian group $A(x)$, with a \v Cech cover $S$. Then for all $n:\N$ we have an exact sequence:
\[H^{n}(X,\lambda x.A(x)^{S(x)}) \to H^{n}(X,L)\to H^{n+1}(X,A)\to 0\]
natural in $A$, where $L(x) = A(x)^{S(x)}/A(x)$.
\end{lemma}

\begin{proof}
  From the long exact cohomology sequence we just need to prove that $H^{n+1}(X,\lambda x.A(x)^{S(x)}) = 0$. This is \Cref{TrivialHasVanishingCohomology}.
\end{proof}
We will continue to show that the above sequence is also exact if we replace the cohomology with the \v Cech cohomology with a \v Cech cover. We need a helper lemma first:
\begin{lemma}\label{long-exact-cech-cohomology-general}
Assume given $X$ and a short exact sequence of abelian groups 
\[0\to A(x)\to B(x)\to C(x)\to 0\]
depending on $x:X$. 
Also assume $S$ a \v Cech cover for $X,A$. 
Then there is a long exact sequence of \v{C}ech cohomology groups:
\[\check{H}^0(X,S,A) \to\check{H}^0(X,S,B) \to\check{H}^0(X,S,C) \to\check{H}^1(X,S,A) \to\check{H}^1(X,S,B) \to\check{H}^1(X,S,C) \to\cdots \]
Moreover this long exact sequence is natural in the short exact sequence.
\end{lemma}

\begin{proof}
%  As $\|K(\Pi_{x:X}A(x)^{S_x^k},1) \|_0= 0$ for all $k\geq 1$, 
  Because of \Cref{TrivialHasVanishingCohomology},
  the short exact sequence induces a short exact sequence of \v Cech complexes. 
  Then we use the long exact cohomology sequence induced by such a short exact sequence of complexes.
\end{proof}

\begin{lemma}\label{inductive-definition-cech-cohomology-general}
Assume given $X$ a type, $A$ a dependent abelian group and $S$ a \v{C}ech cover. 

Then for all $n:\N$ we have an exact sequence:
\[\check{H}^{n}(X,S,\lambda x.\, A(x)^{S(x)}) \to \check{H}^{n}(X,S,L)\to \check{H}^{n+1}(X,S,A)\to 0\]
natural in $A$ where $L(x)=A(x)^{S(x)}/A(x)$.
\end{lemma}

\begin{proof}
By \Cref{long-exact-cech-cohomology-general}, it is enough to prove $\check{H}^{n+1}(X,S,\lambda x.\, A(x)^{S(x)}) = 0$. 
This is \Cref{cech-coefficient-lemma}.
\end{proof}
