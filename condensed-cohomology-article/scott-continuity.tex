The goal of this section is to prove that the axioms of Barton-Commelin's condensed type theory hold in synthetic Stone duality. Sections \ref{stone-spaces} and \ref{overtly-discrete-types} contain generalities on Stone spaces and overtly discrete types. \Cref{scott-continuity-for-cohomology} contains the proof of Barton-Commelin's axioms restricted to Stone spaces, which is the only case required for our cohomology computations. \Cref{barton-commelin} contains the proof of the axioms proper, meaning it extends results from \Cref{scott-continuity-for-cohomology} to compact Hausdorff spaces. It can be skipped by readers only interested in the cohomology results.

\begin{remark}[On notation]
  \item We will use the phrase "tower" to both refer to an $(\N , \geq)$- and $(\N, \leq)$-indexed diagram in some category. The reader can determine which is meant depending on whether we take a limit or colimit. 
  \item For $(C_n)_{n:\N}$ some tower with a limit, we may denote the limit as $C$. We will denote $x\mapsto x_k$ both for the projection maps $C \to C_k$ and any $C_l \to C_k$ with $l\geq k$.
  \item 
  If $(I_n)_{n:\N}$ is some tower with a colimit, we may denote $I$ for the colimit, we may use $i\mapsto [i]$ both for the inclusion maps $I_n \to I$ and $I_n \to I_m$ with $n\leq m$. 
\item If $(C_n(x))_{n:\N}$ is any tower depending on some $x:X$, we may denote $C(x)$ for the limit or colimit if this is unambiguous in context. 
\item If $(p_k:C_k \to D_k)_{k:\N}$ is a map of towers, we may also denote $p:C \to K$ to denote the induced map between limits or colimits. 
\item If $S$ is a set and $P$ an $S$-indexed family of propositions, we may denote $\{s:S | P s\}$ for the $\Sigma$-type $\Sigma_SP$. 
\end{remark} 

\subsection{Stone spaces}
\label{stone-spaces}

We recall generalities about Stone spaces. Most results come from \cite{synthetic-stone-duality}. 

\begin{definition}
A type is a Stone space if it merely is a sequential limit of finite types.

Propositions that are Stone are called Closed
\end{definition}

\begin{remark}
  The definition of Stone spaces above is equivalent to the definition of Stone spaces as spectrum of countably presented booolean algebras given in \cite{synthetic-stone-duality} by Lemmas 3.1 and 3.2 from \cite{synthetic-stone-duality}. We use this alternative definition to emphasis the duality between Stone spaces and overtly discrete types. Indeed overtly discrete types will be defined as sequential colimits of finite types in \Cref{dfnODisc}.
%
  The definition of closed propositions above is equivalent to that of $\mathbb N$-indexed conjunctions of decidable propositions given in \cite{synthetic-stone-duality} by Corollary 3.9 of \cite{synthetic-stone-duality}. 
\end{remark}

\begin{definition}
  We say a class of types $\mathcal C$ has local choice if for any $X:\mathcal C$, and $P$ a family of types over $X$ with $\Pi_{x:X} \| P x\| $, there merely exists some $Y:\mathcal C$ and surjection $q: Y \twoheadrightarrow X $ such that $\| \Pi_{y:Y} P (q y)\|$. 
\end{definition}

\begin{proposition}\label{Stone-closure}
We have the following:

\begin{enumerate}[(i)]
  \item\label{Stone-finite} Finite types are Stone spaces.
  \item\label{Stone-id-Sigma} Stone spaces are stable under identity types and and $\Sigma$-types.
  \item\label{Stone-seqLimits} Stone spaces are stable under sequential limits.
  \item\label{Stone-choice} Stone spaces have local choice
  \item\label{Stone-bounded} Any map from a Stone space into $\N$ is bounded by a maximum value.
\end{enumerate}
\end{proposition}

\begin{proof}
  \ref{Stone-finite} is clear. \ref{Stone-id-Sigma} is Lemma 3.10 and Theorem 4.18 of \cite{synthetic-stone-duality}. \ref{Stone-seqLimits} follows from the observation that countably presented Boolean algebras are closed under sequential colimits and Stone duality. \ref{Stone-choice} and \ref{Stone-bounded} are Axiom 3 and Lemma 3.6 of \cite{synthetic-stone-duality} respectively. 
\end{proof}

Next we give two compactness results for Stone spaces. The first one is used as a definition of compactness in synthetic topology \parencite[Definition 2.5]{lesnik2021synthetictopologyconstructivemetric}, while the second one is closer to the usual topological formulation of compactness.

\begin{lemma}\label{compact-hausdorff-compact}
Assume given a Stone space $S$ and an open subset $U\subseteq S$. Then the proposition $\forall(x:S).\, x\in U$ is open.
\end{lemma}

\begin{proof}
This is Corollary 4.4 of \cite{synthetic-stone-duality}.
\end{proof}

\begin{lemma}\label{compact-hausforff-countable-cover}
Assume given $S:\Stone$ and $(U_i)_{i:\N}$ open subsets of $S$ such that $U_i\subseteq U_{i+1}$ for all $i:\N$. 
If we have that $\bigcup_{i:\N}U_i = S$, then there exists a $k$ such that $U_k=S$.
%If $\forall(x:S).\, \exists(i:\N).\, x\in U_i$ then $\exists(k:\N).\,\forall(x:S).\, x\in U_k$.
\end{lemma}

\begin{proof}
By local choice we have a surjective map $f: T \to S$
such that $\Pi_{x:T} \Sigma_{i:\N}\, f(x)\in U_i$. By boundedness there is $k:\N$ such that $f(x)\in U_k$ for all $x:T$
and then we conclude by surjectivity of $f$.
\end{proof}

Next we give a finite approximation result for surjections between Stone spaces, which will be key in showing the vanishing of the first cohomology groups of Stone spaces.

\begin{lemma}\label{finite-approximation-stone-surjection-cohomology}
Assume given $S:\Stone$ and $T:S\to \Stone$ such that $\Pi_{x:S}\, \propTrunc{T(x)}$. Then there exists a family $(T_k: S\to \Stone)_{k:\N}$ with maps $d_k:\Pi_{x:S}\, T_{k+1}(x)\to T_k(x)$ such that:

\begin{itemize} 
\item For all $x:S$ we have $\mathrm{lim}_k\, T_k(x) = T(x)$.
\item For all $k:\N$ we have $\propTrunc{\Pi_{x:S}\, T_k(x)}$.
\end{itemize}
\end{lemma}

\begin{proof}
  By \Cref{Stone-closure}, the total space $\Sigma_S T$ is Stone. As the projection $\pi: \Sigma_S\, T \to S$ is  assumed surjective, by Remark 3.4 of \cite{synthetic-stone-duality} 
we can write $\pi$ as a sequential limit of surjections $\pi_k : Q_k\twoheadrightarrow S_k$ between finite sets. 
We define $T_k(x)$ as $\pi_k^{-1}(x_k)$ which is indeed a Stone space as it is a closed subset of a finite set (Corollary 3.12 of \cite{synthetic-stone-duality}.
%  
  Since taking fibers and limits commute we get $\lim_k\, T_k(x) = T(x)$.
%  
  Since any $\pi_k$ is a surjection between finite sets we merely have $s_k: \Pi_{x:S_k}\, \pi_k^{-1}(x)$, 
  from which we get $x\mapsto s_k(x_k) : \Pi_{x :S}\, T_k(x)$. 
\end{proof}



\subsection{Overtly discrete types}
\label{overtly-discrete-types}

Here we give generalities on overtly discrete types. Most results come from \cite{synthetic-stone-duality}.

\begin{definition}\label{dfnODisc}
A type is overtly discrete if it merely is a sequential colimit of finite types.
An overtly discrete proposition is called open. 
\end{definition}
\begin{remark}
  The definition of open propositions above is equivalent to that of $\mathbb N$-indexed disjunctions of decidable propositions given in \cite{synthetic-stone-duality} by Lemma 2.8 of \cite{synthetic-stone-duality}. 
\end{remark}

\begin{definition}
A type is countable if it merely is a decidable subset of $\N$.
\end{definition}

\begin{remark}\label{overtly-discrete-countably-presented}
  In Corollary 2.14 of \cite{synthetic-stone-duality}, it is shown that a Boolean algebra is countably presented if and only if it is overtly discrete. Such a result holds if we replace "Boolean algebra" by any finiteary algebraic notion, such as group or Abelian group. The following Lemma gives the intuition that in general overtly discrete types are countably presented sets. 
\end{remark}
\begin{lemma}\label{overtly-discrete-colimit-finite}
Let $X$ be a type, the following are equivalent:

\begin{enumerate}[(i)]
\item $X$ is overtly discrete.
\item $X$ is a quotient of a countable type by an open equivalence relation.
\end{enumerate}
\end{lemma}

\begin{proof}
This is Lemma 2.12 of \cite{synthetic-stone-duality}.
\end{proof}

Now we give the stability properties enjoyed by overtly discrete types. In Barton-Commelin's condensed type theory, a class of types called overtly discrete types enjoying these properties is assumed given.

\begin{proposition}\label{overtly-discrete-closure}
We have the following:

\begin{enumerate}[(i)]
  \item\label{ODisc-finite} Finite types are overtly discrete.
  \item\label{ODisc-id-sigma} Overtly discrete types are stable under identity types and $\Sigma$-types.
  \item\label{ODisc-seqcolim} Overtly discrete type are stable under sequential colimits.
  \item\label{ODisc-quotient} Overtly discrete types are stable under quotients by open equivalence relations.
  \item\label{ODisc-choice} Overtly discrete types have local choice.
\end{enumerate}
\end{proposition}

\begin{proof}
  \ref{ODisc-finite} is clear, \ref{ODisc-id-sigma} and \ref{ODisc-seqcolim} are Lemmas 2.7 and 2.6 of \cite{synthetic-stone-duality} respectively. \ref{ODisc-quotient} follows from \Cref{overtly-discrete-colimit-finite}. For \ref{ODisc-choice}, note that any overtly discrete type is covered by a countable type, which enjoys dependent choice by Axiom 4 of \cite{synthetic-stone-duality}.
\end{proof}

%\end{remark}



\subsection{Scott continuity for Stone spaces}
\label{scott-continuity-for-cohomology}

Here we prove the results required for our cohomology computations, namely Propositions \ref{scott-continuity-right}, \ref{scott-continuity-left} and \ref{tychonov-dual-stone}. We then sum up these results nicely as \Cref{eine-scott-continuity-stone}, which is a dependent version of the following statement: 

\begin{goal}\label{GoalScottGeneralNonDependent}[Non-dependent version]
Given a tower of Stone spaces $(S_k)_{k:\N}$ and a tower of overtly discrete types $(I_j)_{j:\N}$, we have that the canonical map $\mathrm{colim}_{k,j}(S_k\to I_j) \to (\mathrm{lim}_k S_k\to \mathrm{colim}_j I_j)$ is an equivalence.
\end{goal}

\begin{remark}
We call the above result and its generalisations Scott continuity. 
\end{remark}

First we deal with the case where $S_k$ is constant and the target is dependent. A key point is to show any map in $\Pi_{x:S}\, \mathrm{colim}_i\, I_i(x)$ with $S$ Stone and $I_i(x)$ overtly discrete factors through one of the $I_i$. To do this we will show it factors through $\mathrm{Im}(I_i)$ for some $i:\N$ and conclude from there. We start by an auxiliary lemma.

\begin{lemma}\label{overtly-discrete-union-open}
Given a tower of overtly discrete types $(I_i)_{i:\N}$, for any $i:\N$ we have that $\mathrm{Im}(I_i)\subseteq I$ is open. 
\end{lemma}

\begin{proof}
 For any $y:I$, we have that $y\in \mathrm{Im}(I_i)$ is by definition $\propTrunc{\Sigma_{x:I_i}\, x=_{I}y}$,
which is the propositional truncation of an overtly discrete type and therefore open.
\end{proof}

Now we show that factoring through the image is enough.

\begin{lemma}\label{factorisation-image-true-factorisation}
  Assume given a Stone space $S$ and for each $x:S$ a tower of overtly discrete types $(I_i(x))_{i:\N}$. 
  Then the canonical map $ \mathrm{colim}_j\Pi_{x:S} I_j(x) \to 
  \mathrm{colim}_i \Pi_{x:S} (\mathrm{Im} (I_i(x)))$ is surjective. 

  %Given $f:\Pi_{x:S}\,\mathrm{colim}_i\, I_i(x)$ such that there exists some $i:\N$ such that for all $x:S$, $f$ factors through $\mathrm{Im}(I_i(x))$, we have that $f$ factors through $\Pi_{x:X}I_j(x)$ for some $j\geq i$.
\end{lemma}

\begin{proof}
  Let $i:\N, f:\Pi_{x:S}(\mathrm{Im}(I_i(x)))$. By local choice there exists a surjective map $p:T\to S$
with $g:\Pi_{x:T}\, I_i(p(x))$ such that $[g(x)] =_{I(x)} f(p(x))$ for all $x:T$.
Then we have
\[\forall(x,y:T).\, (p(x)=p(y) \to [g(x)] =_{I(x)} [g(y)])\]
so that
\[ \{x,y:T\ |\ p(x)=p(y)\} \to \exists(i:\N). [g(x)] =_{I_i(x)} [g(y)].\]
Since $\{x,y:T\ |\ p(x)=p(y)\}$ is Stone, we can apply \Cref{compact-hausforff-countable-cover} to get a $j:\N$ such that:
\[\forall(x,y:T).\, (p(x)=p(y) \to [g(x)] =_{I_j(x)} [g(y)])\]
which gives a factorisation of $f$ through $I_j$.
\end{proof}

From this, we can conclude \Cref{GoalScottGeneralNonDependent} in the case where the source $S_k$ is constant in $k$. 
\begin{proposition}\label{scott-continuity-right}
Assume given a Stone space $S$ and for each $x:S$ a tower of overtly discrete types $(I_i(x))_{i:\N}$. Then the canonical map $\mathrm{colim}_i (\Pi_{x:S} I_i(x)) \to \Pi_{x:S} \mathrm{colim}_i I(x)$ is an equivalence.
\end{proposition}

\begin{proof}
First we check the canonical map is injective. Assume $f,g:\Pi_{x:S} I_i(x)$ such that
\[\forall(x:S).\, f(x) =_{I(x)} g(x).\]
We have that
\[\forall(x:S).\, \exists(j:\N).\, f(x) =_{I_j(x)} g(x)\]
so by \Cref{compact-hausforff-countable-cover} there exist a $k:\N$ such that
\[\forall(x:S).\, f(x) =_{I_k(x)} g(x).\]
This precisely means that $f=g$ in $\mathrm{colim}_i\, \Pi_{x:S} I_i(x)$. Note that this is sufficient to show injectivity. 

Now we check that the canonical map is surjective. Given a map $f: \Pi_{x:S}\, \mathrm{colim}_i\, I_i(x)$ we know that
\[\forall(x:S).\, \exists(i:\N).\, f(x)\in \mathrm{Im}(I_i(x)).\]
But by \Cref{overtly-discrete-union-open} we have that the proposition $f(x)\in \mathrm{Im}(I_i(x))$ is open, so by \Cref{compact-hausforff-countable-cover} there exists $j:\N$ such that:
\[\forall(x:S).\, f(x)\in \mathrm{Im}(I_j(x))\]
This precisely means that $f$ factors through $\mathrm{Im}(I_j)$. We conclude by \Cref{factorisation-image-true-factorisation}.
\end{proof}

Now we deal with the case where target is constant and non-dependent. If the target is finite this follows from duality.

\begin{lemma}\label{factorisation-stone-finite}
Assume given a tower of Stone spaces $(S_k)_{k:\N}$ and a finite type $F$. Then the canonical map $\mathrm{colim}_k\,(S_k\to F)\to \left(\mathrm{lim}_k\,S_k\to F\right)$ is an equivalence.
\end{lemma}

\begin{proof}
Indeed we have that
\begin{eqnarray}
\mathrm{colim}_k\,(S_k\to F) &=& \mathrm{colim}_k\, \Hom(2^F, 2^{S_k})\nonumber\\
&=& \Hom(2^F, \mathrm{colim}_k\,2^{S_k})\nonumber\\
&=& \Spec(\mathrm{colim}_k\,2^{S_k})\to \Spec(2^F) \nonumber\\
&=& \mathrm{lim}_k\,S_k\to F\nonumber
\end{eqnarray}
where the second line comes from the fact that $2^F$ is finitely presented, and the other lines follow from Stone duality. We omit the verification that this is indeed the canonical map.
\end{proof}

Now we extend to the case where the target is a constant arbitrary overtly discrete type.

\begin{proposition}\label{scott-continuity-left}
Assume given a tower of Stone spaces $(S_k)_{k:\N}$ and an overtly discrete type $I$. Then canonical map $\mathrm{colim}_k\,(S_k\to I)\to \left(\mathrm{lim}_k\,S_k\to I\right)$ is an equivalence.
\end{proposition}

\begin{proof}
There exists a tower of finite types $(I_i)_{i:\N}$ so that $I = \mathrm{colim}_i\,I_i$. By \Cref{scott-continuity-right} together with \Cref{factorisation-stone-finite} we have that
\begin{eqnarray}
\mathrm{colim}_k\,(S_k\to \mathrm{colim}_i\,I_i) &=& \mathrm{colim}_{k,i}\,(S_k\to I_i)\nonumber\\
&=& \mathrm{colim}_i\,(\mathrm{lim}_k\,S_k\to I_i)\nonumber\\
&=& \mathrm{lim}_k\,S_k\to \mathrm{colim}_i\,I_i.\nonumber
\end{eqnarray}
We omit the verification that this is indeed the canonical map.
\end{proof}

Now we prove the dual of Tychonov for Stone spaces, i.e.\ that a product of overtly discrete types indexed by a Stone space is itself overtly discrete. We need an auxiliary lemma.

\begin{lemma}\label{tychonov-dual-auxiliary}
Assume given a surjection between Stone spaces $p:T\to S$ and for each $x:S$ an overtly discrete type $I(x)$. If $\Pi_{x:T}I(p(x))$ is overtly discrete then so is $\Pi_{x:S}I(x)$.
\end{lemma}

\begin{proof}
Since the map is surjective we have an embedding $\Pi_{x:S}I(x)\subset \Pi_{x:T}I(p(x))$. But the fiber over $g:\Pi_{x:T}I(p(x))$ is
\[\forall (x,y:T).\, p(x)=p(y) \to g(x)=g(y)\]
which is open by \Cref{compact-hausdorff-compact}. Therefore $\Pi_{x:S}I(x)$ is an open subtype of an overtly discrete type, so it is itself overtly discrete.
\end{proof}

\begin{proposition}[Tychonov's Dual]\label{tychonov-dual-stone}
Assume given a Stone space $S$ and for each $x:S$ an overtly discrete type $I(x)$. Then $\Pi_{x:S}I(x)$ is overtly discrete.
\end{proposition}

\begin{proof}
By local choice and \Cref{tychonov-dual-auxiliary}, it is sufficient to show that the above holds if we assume given for each $x:S$ a tower of finite types $(\mathrm{Fin}(l_{k,x}))_{k:\N}$
such that for all $x:S$ we have that $I(x) = \mathrm{colim}_k\, \mathrm{Fin}(l_{k,x})$.
Then using \Cref{scott-continuity-right} and the fact that overtly discrete types are stable under sequential colimits, it is enough to prove that $\Pi_{x:S} \mathrm{Fin}(l_{k,x})$ is overtly discrete for any $k:\mathbb N$. 

Using boundedness there exists some $n:\N$ such that $\Pi_{x:S} \mathrm{Fin}(l_{k,x}) = \Pi_{i<n}(S_i\to \mathrm{Fin}(i))$ where $S_i = \{x:S\ |\ l_{k,x} = i\}$. Then by \Cref{factorisation-stone-finite} we know that each $S_i\to \mathrm{Fin}(i)$ is overtly discrete, and we conclude using the fact that overtly discrete types are closed under finite products.
\end{proof}

So far we have two versions of Scott continuity (Propositions \ref{scott-continuity-right} and \ref{scott-continuity-left}), neither clearly implying the other. They are sufficient to derive our cohomology results. In the rest of this section we will prove a common generalisation suggested to us by Reid Barton. We start by an auxiliary definition.

\begin{definition}\label{category-scott-continuity}
We have a category ${\mathcal C}_\Stone$ defined by:

\begin{itemize}
\item An object consists of $S:\Stone$ and $I:S\to \ODisc$.
\item A morphism from $(S,I)$ to $(T,J)$ consists of $p:T\to S$ with $q:\Pi_{x:T}(I(p(x))\to J(x))$.
\end{itemize}
\end{definition}

The arrows are defined such that we have a covariant functor $\Pi:{\mathcal C}_\Stone\to \ODisc$. It is indeed valued in $\ODisc$ by \Cref{tychonov-dual-stone}. Now we prove the general version of Scott continuity for Stone spaces.

\begin{theorem}[Scott continuity for Stone spaces]\label{eine-scott-continuity-stone}
The functor $\Pi : {\mathcal C}_\Stone \to \ODisc$ commutes with sequential colimits.
\end{theorem}

&=& \mathrm{colim}_i\,\Pi_{y:\mathrm{lim}_{k}S_k}\, I_i(y_i)\nonumber\\
&=& \Pi_{y:\mathrm{lim}_{k}S_k}\, \mathrm{colim}_i\, I_i(y_i).\nonumber
\end{eqnarray}

\subsection{Barton-Commelin's condensed type theory axioms}
\label{barton-commelin}

We already have proven all the results needed for our cohomology computations. In this section complete the proof that Barton-Commelin's condensed type theory axioms hold in synthetic Stone duality. These axioms postulate two classes of types (called $\CHaus$ and $\ODisc$) such that:
\begin{itemize}
\item Both classes are closed under identity types, $\Sigma$-types and quotients. 
\item $\Chaus$ is closed under sequential limits and $\ODisc$ under sequential colimits.
\item Both classes satisfy local choice.
\item The classes interact well in the sense that they satisfy Tychonov, its dual and Scott continuity.
\end{itemize}

We have already defined the class of overtly discrete types and proved its required closure properties in \Cref{overtly-discrete-closure}. We did show that Stone spaces enjoy the required interaction results with overtly discrete types in \Cref{scott-continuity-for-cohomology}. But they are not closed under quotient. The solution is to define compact Hausdorff spaces as quotients of Stone spaces by closed equivalence relation, and then extend the interaction results from Stone spaces to compact Hausdorff spaces.

\begin{definition}
A type $C$ is a compact Hausdorff space of its identity types are closed and there exists a Stone space $S$ with a surjection $S\to C$.
\end{definition}

To prove that compact Hausdorff spaces are closed under sequential limits, we need the following auxiliary lemma.

\begin{lemma}\label{sequential-limit-Hausdorff}
Assume given a tower of compact Hausdorff spaces $(C_k)_{k:\N}$. There exists a tower of Stone spaces $(S_k)_{k:\N}$ with a map of towers $(S_k\to C_k)_{k:\N}$ which is level-wise surjective, such that the induced map $\mathrm{lim}_k\,S_k\to \mathrm{lim}_k\,C_k$ is also surjective.

%\begin{center}
%\begin{tikzcd}
%\cdots \ar[r]& S_1\ar[r]\ar[d] & S_0 \ar[d]\\
%\cdots \ar[r] & C_1\ar[r] & C_0 \\
%\end{tikzcd}
%\end{center}

%such that the map:
%\[S_0\to C_0\]
%is surjective and for all $n:\N$ the induced map:
%\[S_{n+1} \to C_{n+1}\times_{C_n} S_n\]
%is surjective.

%This implies that the induced map:
%\[\lim_kS_k \to \lim_k C_k\]
%is surjective.
\end{lemma}

\begin{proof}
We will build a tower of Stone spaces $(S_k)_{k:\N}$ and a map of towers $(p_k:S_k\to C_k)_{k:\N}$ such that the induced maps $S_{k+1}\to C_{k+1}\times_{C_k}S_k$ are surjective, which immediately implies $p_k$ level-wise surjective by induction on $k$, and implies the induced map $\mathrm{lim}_k\,S_k\to \mathrm{lim}_k\,C_k$ surjective by dependent choice.

For $k=0$, by definition there exists a Stone space $S_0$ with a surjection $p_0:S_0\to C_0$. 

Next assume a surjection $p_k:S_k\to C_k$. We shall show there merely exists a Stone space $S_{k+1}$ with a map $p_{k+1}:S_{k+1}\to C_{k+1}$ making the appropriate square commute, such that the induced map $S_{k+1}\to C_{k+1}\times_{C_k} S_k$ is a surjection (and therefore $p_{k+1}$ is a surjection).
%Using dependent choice, it is enough to show that we can merely extend such a tower $(S_k)_{k\leq n}$ to $(S_k)_{k\leq n+1}$.

By definition, there exists a Stone space $T$ with a surjection $q:T \to C_{k+1}$. Then we define $S_{k+1}$ such that the following squares are pullback:
\begin{center}
\begin{tikzcd}
S_{k+1}\ar[d,swap,"\pi_1"]\ar[r] & C_{k+1}\times_{C_k} S_k\ar[r]  \ar[d]& S_k\ar[d,"p_k"]\\
T \ar[r,swap,"q"] & C_{k+1}\ar[r]  & C_k
\end{tikzcd}
\end{center}
Note that $S_{k+1}$ is Stone by \Cref{Stone-closure}.
We define $p_{k+1} = q\circ \pi_1$. The induced map in $S_{k+1}\to C_{k+1}\times_{C_k} S_k$ is a surjection as it is a pullback of the surjection $q$. 

By dependent choice we get the required map of towers and surjectivity of the induced map $\lim_kS_k \to \lim_kC_k$. 
\end{proof}

We are now ready to prove the required closure properties for compact Hausdorff spaces:

\begin{proposition}
We have the following:

\begin{enumerate}[(i)]
  \item\label{CHaus-finite} Finite types are compact Hausdorff spaces.
  \item\label{CHaus-id-sigma} Compact Hausdorff spaces are stable under identity types and and $\Sigma$-types.
  \item\label{CHaus-quotient} Compact Hausdorff spaces are stable under quotients by closed equivalence relations.
  \item\label{CHaus-seqlim} Compact Hausdorff spaces are stable under sequential limits.
  \item\label{CHaus-choice} Compact Hausdorff space have local choice.
\end{enumerate}
\end{proposition}

\begin{proof}
  Note that all Stone types are compact Hausdorff, and identity types of Compact Hausdorff types are Stone. Thus \ref{CHaus-finite}, \ref{CHaus-quotient}, \ref{CHaus-choice} and closure under identity types follow from \Cref{Stone-closure}. \ref{CHaus-id-sigma} then follows from Lemma 4.11 of \cite{synthetic-stone-duality}. For \ref{CHaus-seqlim}, 
assume given a tower of compact Hausdorff spaces $(C_k)_{k:\N}$. Closed proposition are stable under sequential limits, so identity types in $\mathrm{lim}_k\,C_k$ are closed. By \cref{sequential-limit-Hausdorff} we get a surjection from a Stone space to $\mathrm{lim}_k\,C_k$,
\end{proof}

Next we need to prove Tychonov and its dual for compact Hausdorff spaces.

\begin{proposition}[Tychonov]
Assume given an overtly discrete type $I$ and for each $i:I$ a compact Hausdorff space $C(i)$. Then $\Pi_{i:I}\,C(i)$ is a compact Hausdorff space.
\end{proposition}

\begin{proof}
We can assume $I = \mathrm{colim}_{k:\N}\,I_k$ with $I_k$ finite. Then $\Pi_{i:I}\,C(i) = \lim_{k:\N}\, \Pi_{j:I_k}\,C([j])$
and we can conclude using that compact Hausdorff spaces are stable under sequential limits and finite products.
\end{proof}

Extending the Tychonov's dual from Stone spaces to compact Hausdorff spaces only requires generic reasoning on quotients.

\begin{proposition}[Tychonov's dual]
Assume given a compact Hausdorff space $C$ and for each $x:X$ an overtly discrete type $I(x)$. Then $\Pi_{x:C}\,I(x)$ is an overtly discrete type.
\end{proposition}

\begin{proof}
We consider a surjection $p:S\to C$ with $S$ Stone. Then
\[\Pi_{x:C}\,I(x) = \{f:\Pi_{x:S}\,I(p(x))\ |\ \forall(x,y:S).(p(x)=p(y)\to f(x)=f(y))\}.\]
But $\forall(x,y:S).(p(x)=p(y)\to f(x)=f(y))$ is open by \Cref{compact-hausdorff-compact} and $\Pi_{x:S}\,I(p(x))$ is overtly discrete by \Cref{tychonov-dual-stone}, so we can conclude.
\end{proof}

Finally we extend Scott continuity to compact Hausdorff spaces. We write ${\mathcal C}_\CHaus$ for ${\mathcal C}_\Stone$ as given in \Cref{category-scott-continuity} with Stone spaces replaced by compact Hausdorff spaces. 

\begin{theorem}[Scott Continuity]\label{scott-continuity-chaus}
The functor $\Pi : {\mathcal C}_\CHaus \to \ODisc$ commutes with sequential colimits.
\end{theorem}

\begin{proof}
Assume given a tower in ${\mathcal C}_\Chaus$ denoted by $(C_i,I_i)_{i:\N}$. By \Cref{sequential-limit-Hausdorff} we can consider a tower of Stone spaces $(S_i)_{i:\N}$ with a level-wise surjection $(p_i:S_i\to C_i)_{i:\N}$ to the tower $(C_i)_{i:\N}$.% We write $S=\mathrm{lim}_i\,S_i$ and $C=\mathrm{lim}_i\,C_i$, for each $x:C$ we write $I(x) = \mathrm{colim}_i\,I_i(x_i)$ and finally we write $p =\mathrm{lim}_i\,p_i$. 
Then the canonical map
\[\mathrm{colim}_i(\Pi_{x:C_i}\,I_i(x) ) \to \Pi_{x:C}\,I(x)\]
is equal to the canonical map
%
\[ \left\{ [f_i] : \mathrm{colim}_i\, \Pi_{x:S_i}\, I_i(p_i(x))\ |\ \mathrm{colim}_{j\geq i}\, Q_j(f_i)\right\} \longrightarrow \left\{f:\Pi_{x:S}\, I(p(x))\ |\ Q(f)\right\} \]
where we pattern-matched on the terms of the colimit and defined
\begin{eqnarray}
  Q_j(f_i) &=& \forall(x,y:S_j).\, p_j(x)=p_j(y)\to [f_i(x_i)]=_{I_j}[f_i(y_i)]\nonumber\\
Q(f) &=& \forall(x,y:S).\, p(x)=p(y)\to f(x)=f(y).\nonumber
\end{eqnarray}
Then by \Cref{eine-scott-continuity-stone} we have that the canonical map
\[\epsilon : \mathrm{colim}_i(\Pi_{x:S_i}\,I_i(p_i(x))) \to \Pi_{x:S}\, I(p(x))\]
is an equivalence, and by \Cref{eine-scott-continuity-stone} again we have that the canonical map
\[\mathrm{colim}_{j\geq i}\,Q_j(f_i) \to Q(\epsilon[f_i])\]
is an equivalence as well so we can conclude.
\end{proof}
