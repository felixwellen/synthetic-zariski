\begin{lemma}\label{ProductTopology}
  Let $X,Y:\CHaus$. Any open of $X\times Y$ can be written as $\bigcup_{i:I} U_i \times V_i$ where $I$ is some overtly discrete set, and for each $i:I$, the sets $U_i,V_i$ are open subsets of $X,Y$ respectively. 
\end{lemma}
\begin{corollary}\label{countableProductTopology}
  Let $X,Y:\CHaus$. Any open of $X\times Y$ can be written as $\bigcup_{n:\N} U_n \times V_n$ where for each $n:\N$, the sets $U_n,V_n$ are open subsets of $X,Y$ respectively. 
\end{corollary}

\begin{lemma}\label{subcoverN} Let $X:\CHaus$ and $U:\N \to X^\Open$ be such that $\bigcup_{n:\N} U_n = X$. 
  Then there exists some finite set $I\subseteq \N$ such that $\bigcup_{i:I} U_i = X$. 
\end{lemma}
\begin{corollary}\label{subcover}
  Let $X:\CHaus$ and $U:J \to X^\Open$ be such that $\bigcup_{j:J} U_j = X$. Then there exists some finite set $I$ and map $\iota:I \to J$ such that $X = \bigcup_{i:I} U_{\iota(i)}$. 
\end{corollary}

\begin{lemma}\label{seperation}
  If $X:\CHaus$ and $C\subseteq D \subseteq X$ for $C$ closed and $D$ open. 
  Then there exists $U,K\subseteq X$ with $U$ open and $K$ closed such that $ C\subseteq U \subseteq K \subseteq D$. 
\end{lemma}
\begin{proof}
  As $C\subseteq D$, we have $\neg D, C$ disjoint and closed. By Lemma 4.9 of \cite{synthetic-stone-duality}, we have disjoint open subsets $U,V$ with $\neg D \subseteq V , C \subseteq U$. Then for $K = \neg V$ we have $U\subseteq K$ and $K \subseteq \neg \neg D$. As closed propostions are double-negation stable, we conclude $K\subseteq D$. 
\end{proof}

\begin{lemma}\label{SubsetUnion}
  Let $X:\CHaus$ and $A, B \subseteq X$ subsets with $A$ open, $B$ closed, or the other way around. 
  Then $X = \neg A \cup B$ if and only if $A \subseteq B$. 
\end{lemma}
\begin{proof}
  Assume $A$ open, $B$ closed, the other proof is similar. Then for all $x:X$ we have $\neg A (x) \vee B(x)$ is $\neg\neg$-stable, hence
  $$(\neg A (x) \vee B (x)) \leftrightarrow (A(x) \to B(x)).$$
\end{proof}
