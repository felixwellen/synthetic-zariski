In this section, we present a proof that the entourages of a compact Hausdorff space form an uniformity in the classical sense. The proof we present here is a refactored version of a proof suggested by claude.

\begin{definition}
  Let $(U_i)_{i:I}$ be an collection of subsets of some space $X$. 
  We define $\widetilde U \subseteq X \times X$ as 
  $\widetilde U = \bigcup_{i:I} U_i \times U_i$. 
\end{definition}
\begin{remark}
  Note that for a collection of subsets $(U_i)_{i:I}$ of a space $X$, we have $\bigcup_{i:I} U_i = X$ iff $\Delta\subseteq \widetilde U$. 
\end{remark}

\begin{lemma}\label{openSquares}
  Let $X:\CHaus$ and $U\subseteq X \times X$ open such that $\Delta\subseteq U$. Then there exists some finite set of opens $(P_i\subseteq X)_{i:I}$ such that $\Delta \subseteq \widetilde P \subseteq U$.
\end{lemma}
\begin{proof}
  By \Cref{ProductTopology}, there exists opens $(V_j)_{j:J},(W_j)_{j:J}$ indexed by some overtly discrete set $J$ such that $U=\bigcup_{j:J} (V_j \times W_j)$. As $\Delta \subseteq U$ we have for each $x:X$ some $j:J$ such that $x\in W_j ,V_j$. 
  It follows that for $P_j = V_j \cap W_j$ that $\Delta \subseteq \widetilde P \subseteq U$. 
  Then $P$ covers $X$ by opens and by \Cref{subcover} has a finite subcover as required. 
\end{proof}

\begin{definition}
  Let $X$ be a type and $(U_i)_{i:I},(V_j)_{j:J}$ be covers of $X$. 
  We say that $U$ $\Delta$-refines $V$ iff for every $x:X$, there is some $j:J$ such that 
  $\bigcup\{U_i | x \in U_i\}  \subseteq V_j$. 
\end{definition}
\begin{lemma}
  For $U,V$ as above, if $U$ $\Delta$-refines $V$, then $\tilde U\circ \tilde U \subseteq \tilde V$. 
\end{lemma}
\begin{proof}
  Let $(x,y), (y,z) \in \widetilde U$. 
  Then there exist $i,k:I$ such that $x,y\in U_i$ and $y,z\in U_k$. 
  Therefore $(U_i \cap U_k)(y)$ and hence there is some $j:J$ such that $U_i \cup U_k \subseteq V_j$. 
  Thus $x, z \in V_j$ and hence $(x,z) \in \widetilde V$. 
\end{proof} 


\begin{lemma}\label{ClosedRefinement}
  Let $X:\CHaus$ and $(U_i)_{i:I}$ be a finite open cover of $X$. Then there exists some $(D_i)_{i:I}$ closed cover of $X$ such that $D_i \subseteq U_i$. 
\end{lemma}
\begin{proof}
  Let $(U_i)_{i:I}$ be a finite open cover. 
  Assume $I = \mathbb N_{\leq n}$. 
  We will recursively define a sequence $C_i \subseteq V_i \subseteq D_i \subseteq U_i$
  with $C_i,D_i$ closed and $V_i$ open, such that for all $l:\mathbb N_{\leq n+1}$ we have 
  \begin{equation}\label{inductionLoop}\bigcup_{i < l} V_i \cup \bigcup_{i\geq l} U_i = X.\end{equation}
  where we intepret $U_{n+1} = \emptyset$. Note that \Cref{inductionLoop} is satsified for $l=0$, and if were to be satisfied for $l=n+1$, this gives that $\bigcup_{i:I} V_i = X$, and hence we would have $\bigcup_{i:I} D_i = X$ as required. 
%
  Assume $C_i,V_i,D_i$ have been defined for $i<k$ and be such that \Cref{inductionLoop} is satisfied for $l<k$. 
  Note that $(\bigcup_{i<k} V_i \cup \bigcup_{i>k} U_i)$ is open, and hence $\neg\neg$-stable, and define $C_k$ as its complement. By the induction hypothesis we have $C_k \subseteq P_k$. By \Cref{seperation}, we find $V_k,D_k$ open and closed respectively such that 
  $C_k \subseteq V_k \subseteq D_k \subseteq U_k$. 
  Now we have that 
  $$
  \bigcup_{i< k} V_i \cup \bigcup_{i\geq k} U_i = 
  (\bigcup_{i<k}V_i \cup 
  \bigcup_{i>k} U_i)
  \cup V_k 
  = 
  \neg C_k \cup V_k,
  $$
  which equals $X$ by \Cref{SubsetUnion}. 
  Thus \Cref{inductionLoop} holds for $l=k$, and we conclude by strong induction. 
\end{proof}
