\section{The topologist's sine curve}
The topologist's sine curve $\mathcal S$ is an example of a topological space that is not path-connected but which is connected. In this section, we shall define $\mathcal S$ in the setting of SSD and recover these results. 
In \Cref{curveShape} we shall show $\mathcal S$ is $[0,1]$-contractible, hence by \Cref{contractibleVSconnected} also connected. 
In \Cref{curveNotPathConnected} we shall recall a proof that $\mathcal S$ is not path-connected in the setting of SSD, and hence not path-contractible. 
It follows that the notions of being path-contractible and $[0,1]$-contractible don't agree. 


Because of \Cref{CauchyDfns} we can make the following definition:
\begin{definition}%[The topologist's sine curve]
  For $k:\mathbb N_{\geq1}$, define $s_k:[\frac1k,1] \to [-1,1]$ by $s_k(x) = \sin(\frac1x)$ and $\mathcal S_k \subseteq [0,1] \times [-1,1]$ by 
  $$\mathcal S_k = \left([0, \frac1k] \times [-1,1]\right) \cup \mathrm{graph}(s_k).$$
  We define \textbf{the topologist's sine curve} $\mathcal S = \bigcap_{k:\mathbb N_{\geq1}} \mathcal S_k$.

\end{definition}
  We shall write $\iota_k : \mathcal S_k \to [0,1] \times [-1,1]$ 
  and $\iota:\mathcal S \to [0,1] \times [-1,1]$ for the inclusion maps, and $\pi_0:[0,1] \times[-1,1] \to [0,1],
\pi_1:[0,1] \times[-1,1] \to [-1,1]$ for the projection maps. 


\begin{lemma}
  The topologist's sine curve is a compact Hausdorff space.
\end{lemma}
\begin{proof}
  Note that both $[0,\frac1k] \times [-1,1]$ and $\mathrm{graph}(s_k)$ are images of compact Hausdorff spaces, hence closed by \Cref{CHausClosedImage}.
  By \Cref{ClosureClosedOpen} it follows that $\mathcal S$ is closed. We conclude by \Cref{CHausClosedImage}.
\end{proof}

\begin{lemma}\label{fiberInhabited}
  $\pi_0\circ \iota : \mathcal S \to [0,1]$ is surjective.
\end{lemma}
\begin{proof}
  We shall show that for each $r:[0,1]$ the fiber $(\pi_0\circ \iota)^{-1}(r)$ is merely inhabited. 
%

  it's sufficient to show that $\pi_0^{-1}(r) \cap \mathcal S_k$
  is merely inhabited for any $k:\mathbb N_{\geq 1}$. 
%
  Let $k:\mathbb N_{\geq 1}$. 
  As we're showing a proposition, we can make a case distinction based on \Cref{LLPO}:
%  Note that $r\leq \frac1k \vee r \geq \frac 1k$. 
  \begin{itemize}
    \item If $r\leq \frac1k$, we have that $(r,0) \in \mathcal S_k$. 
    \item If $r\geq \frac1k$, $\frac1r$ is well-defined and 
      $(r,\sin(\frac1r)) \in \mathcal S_k$. 
  \end{itemize}
  Hence in both cases we have that $\pi_0^{-1}(r) \cap \mathcal S_k$ is inhabited, as required. 
\end{proof}

\begin{lemma}
  For each $k:\mathbb N_{\geq 1}$ and $r:[0,1]$, $\mathcal S_k\cap \pi_0^{-1}(r)$ is convex.
\end{lemma}
\begin{proof}
  Suppose that $(x,y), (x',y')$ both lie in $\mathcal S_k\cap \pi_0^{-1}(r)$.
  Suppose furthermore $\alpha,\beta:[0,1]$ with $\alpha+ \beta = 1$. 
  We will show that $\alpha\cdot (x,y) + \beta\cdot (x',y') $ also lies in
  $\mathcal S_k \cap \pi_0^{-1}(r)$.

  Recall that 
  $\mathcal S_k = [0,\frac1k] \times [-1,1]  \cup \mathrm{graph}(s_k)$. 
  As we will show a proposition, we can make a case distinction 
  for both $(x,y), (x',y')$ on whether they come from $[0,\frac 1k] \times [-1,1]$ or 
  from $\mathrm{graph}(s_k)$.
  \begin{itemize}
    \item 
      If either comes from $[0,\frac1k] \times [-1,1]$, we must have that
      $r \leq \frac1k$.
      Then $\pi_0^{-1}(r) = \{r\} \times [-1,1]\subseteq \mathcal S_k$.
      It follows that $\mathcal S_k \cap \pi_0^{-1}(r) = \{r\} \times [-1, 1]$, which is convex. 
    \item 
      If both come from $\mathrm{graph}(s_k)$,
      then $x = x' = r$ and $y = y' = \sin(\frac1r)$. 
      Hence $(x,y) = (x',y')$ and thus $\alpha(x,y) + \beta(x',y') = (x',y')$. 
  \end{itemize}
  In both cases we conclude that $\alpha (x,y) + \beta(x',y')$ lies in
  $\mathcal S_k\cap \pi_0^{-1}(r)$.
\end{proof}

\begin{corollary}\label{fiberConvex}
  For each $r:[0,1]$, the fiber 
  $(\pi_0\circ\iota)^{-1}(r)$
  %$S \cap (\{r\} \times [-1,1])$ 
  is convex. 
\end{corollary}
\begin{proof}
  Any intersection of convex subspaces of $\mathbb R^n$ is itself convex.
\end{proof}

\subsection{The curve is not path-connected}\label{curveNotPathConnected}
\begin{remark}\label{topbottomSeq}
  For each $k:\N$, denote $t_k = \frac1{(2k+\frac12) \pi}$ and $b_k = \frac1{(2k+\frac32) \pi}$. 
  Note that for all $k:\N$ we have $\sin(\frac1{b_k})=-1, \sin(\frac1{t_k})=1$ and
  $1>t_k>b_k>t_{k+1}>b_{k+1}>0$. 
\end{remark}
\begin{lemma}\label{cofinaltbseq}
  Given $r:[0,1]$ with $r>0$, there exists some $k:\N$ with $t_k,b_k<r$.  
\end{lemma}
\begin{proof}
  By \Cref{positiveNumbersCharacter}, there is some $k:\N$ with $\frac1k<r$. As $(2k+\frac32)\pi,(2k+\frac12)\pi>k$, this $k$ suffices.
\end{proof}
\begin{lemma}\label{NonZeroOnGraph}
  Let $(x,y):[0,1]\times[-1,1]$ be such that $(x,y)\in\mathcal S$ and $x>0$.
  Then $y = \sin(\frac1x)$.
\end{lemma}
\begin{proof}
  By \Cref{positiveNumbersCharacter}, there is some $k:\N$ such that $\frac1k<x$ and hence $(x,y)\in \mathcal S_k$ and $y = \sin(\frac1x)$. 
\end{proof}

\begin{lemma}\label{SineWillHitTopBot}
  Let $\gamma:[0,1] \to \mathcal S$. 
  Assume that $\pi_0(\gamma(0)) = 0$ and $\pi_0(\gamma(r))>0$ for some $r:[0,1]$.
  Then there exist some $u,v:[0,1]$ with $0<u<v<r$ such that $\pi_1(\gamma(u)) = 1$ and $\pi_1(\gamma(v)) = -1$.
\end{lemma}
\begin{proof}
  By \Cref{cofinaltbseq}, there exists some $k:\N$ such that $0 < b_k<\pi_0(\gamma(r))$. 
  We can thus apply \Cref{IntermediateValue} to $\pi_0\circ \gamma : [0,r] \to [0,1]$ to find some $v:[0,r]$ with $\pi_0(\gamma(v)) = b_k$. Then by \Cref{NonZeroOnGraph} and \Cref{topbottomSeq} we find that $\pi_1(\gamma(v)) = -1$.
  The case for $u$ is similar. 
\end{proof}

\begin{theorem}[The topologist's sine curve is not path-connected]
  There is no map $\gamma:[0,1] \to \mathcal S$ such that $\gamma(0) = (0,0)$ and $\gamma(1) = (1,\sin(1))$. 
\end{theorem}
\begin{proof}
  By \Cref{UniformContinuity} applied to 
  $\pi_1\circ \iota \circ \gamma:[0,1] \to [-1,1]$, for $\epsilon = 2$, 
  there exists some $\delta:\R_{>0}$ such that for any $u,v:[0,1]$ with 
  $|\pi_1(\iota(\gamma(u)))- \pi_1(\iota(\gamma(v)))| = 2$ we have $|u-v|\geq \delta$. 
  Let $N:\N$ be such that $\frac1N>\frac\delta2$. 
  Applying \Cref{SineWillHitTopBot} $N$ times, we can find a sequence 
  $u_i,v_i,~i:\N_{\leq N}$ such that $\pi_1(\gamma(u_i)) = 1, \pi_1(\gamma(v_i) = -1$ and $u_{i+1},v_{i+1}<u_i<v_i$ for all $i:\mathbb N_{<n}$. 
  But then the sequence $u_i$ is decreasing with steps at least $2\delta>0$, and thus $u_N<0$, while $u_N:[0,1]$, which is a contradiction.

\begin{remark}
  One could define an equivalence relation on any type $X$ such that 
  $x\sim y$ iff there is a path from $x$ to $y$. 
  The equivalence classes of this relation would be called the path-connected components. 
  We conjecture that the space of path-connected components of $\mathcal S$ is equivalent to the type of open propositions, which is the type of propositions $P$ for which there exists some $x:[0,1]$ such that $P\leftrightarrow x>0$.
\end{remark}
